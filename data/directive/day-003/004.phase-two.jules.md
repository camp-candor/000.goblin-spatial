#### 2. Receipt Minting Buzzer: `packages/301.spatial/42.receipt.unit/buz/receipt.buzz.ts`

```typescript
import type { PerceptCandidate, PerceptEvidence, AuditQuadruple, AxialCoord } from '../fce/receipt.interface.js';

export interface MintPerceptParams {
  readonly sourceCoord: AxialCoord;
  readonly observerCoord: AxialCoord;
  readonly qualityScorePerMille: number;
  readonly arrivalBearing: number | null;
  readonly isOccluded: boolean;
  readonly evidence: PerceptEvidence;
  readonly auditQuadruple: AuditQuadruple;
}

export const mintPerceptReceipt = (model: any, bal: any): any => {
  const params: MintPerceptParams = bal?.dat?.perceptCandidate || bal?.perceptCandidate;

  if (!params || !params.evidence || !params.auditQuadruple) {
    if (bal?.slv) {
      bal.slv({ rcpBit: { idx: 'mint-receipt-err', val: -1, dat: { error: 'INVALID_PERCEPT_PARAMS' } } });
    }
    return model;
  }

  const candidate: PerceptCandidate = Object.freeze({
    sourceCoord: Object.freeze({ ...params.sourceCoord }),
    observerCoord: Object.freeze({ ...params.observerCoord }),
    qualityScorePerMille: params.qualityScorePerMille,
    arrivalBearing: params.arrivalBearing,
    isOccluded: params.isOccluded,
    evidence: Object.freeze({ ...params.evidence }),
    auditQuadruple: Object.freeze({ ...params.auditQuadruple }),
  });

  if (bal?.slv) {
    bal.slv({
      rcpBit: {
        idx: 'mint-receipt-success',
        val: 1,
        dat: { perceptCandidate: candidate },
      },
    });
  }

  return model;
};

```

#### 3. Unit Verification Suite: `packages/301.spatial/42.receipt.unit/test/receipt.test.ts`

```typescript
import { describe, it, expect } from 'vitest';
import { mintPerceptReceipt } from '../buz/receipt.buzz.js';
import type { PerceptCandidate } from '../fce/receipt.interface.js';

describe('301.spatial - [Task 1.2.2] Package Acoustic & Optical Raycast Proofs', () => {
  const AUDIT_QUADRUPLE = {
    topologyHash: 'topo_abc123',
    overlayHash: 'overlay_def456',
    queryHash: 'query_789xyz',
    kernelVersion: '0.1.4',
  };

  it('aggregates acoustic wavefront proof into a sealed, frozen PerceptCandidate receipt', () => {
    let captured: PerceptCandidate | null = null;
    const bale = {
      idx: 'receipt-audit-acoustic',
      dat: {
        perceptCandidate: {
          sourceCoord: { q: 0, r: 0, s: 0 },
          observerCoord: { q: 0, r: 3, s: -3 },
          qualityScorePerMille: 700,
          arrivalBearing: 3,
          isOccluded: false,
          evidence: {
            modality: 'ACOUSTIC',
            intensityCdb: 5000,
            noiseFloorCdb: 2700,
            snrMarginCdb: 2300,
            portalHopCount: 2,
          },
          auditQuadruple: AUDIT_QUADRUPLE,
        },
      },
      slv: (res: any) => {
        captured = res.rcpBit.dat?.perceptCandidate ?? null;
      },
    };

    mintPerceptReceipt({}, bale);

    expect(captured).not.toBeNull();
    expect(captured?.evidence.modality).toBe('ACOUSTIC');
    if (captured?.evidence.modality === 'ACOUSTIC') {
      expect(captured.evidence.snrMarginCdb).toBe(2300);
      expect(captured.evidence.intensityCdb).toBe(5000);
    }
    expect(Object.isFrozen(captured)).toBe(true);
  });
});

```
