 * :: BRANDED INTEGER PRIMITIVES & DOMAIN INTERFACES (01.lattice.unit)

 * Defines nominal branded types, exact integer lattice coordinate contracts,
 * and doubled-coordinate geometric structures for pointy-top hexes.

declare const __brand: unique symbol;

/**
 * Nominal branded type representing an unsigned 32-bit packed axial cell coordinate.
 * Packing schema: ((((q + 32768) & 0xFFFF) << 16) | ((r + 32768) & 0xFFFF)) >>> 0
 */
export type TacticalCellId = number & { readonly [__brand]: 'TacticalCellId' };

/**
 * Nominal branded type representing an unsigned 64-bit commutative packed edge key.
 * Packing schema: (min(cellA, cellB) << 32n) | max(cellA, cellB)
 */
export type TacticalEdgeId = bigint & { readonly [__brand]: 'TacticalEdgeId' };

/**
 * Canonical cubic axial coordinate descriptor.
 * Invariant: q, r, s must be exact integers and satisfy q + r + s === 0.
 */
export interface AxialCoord {
    readonly q: number;
    readonly r: number;
    readonly s: number;
}

/**
 * Exact integer point in doubled coordinate affine space (4q + 2r, 6r).
 * Uses BigInt to ensure 2D cross-products never overflow standard numbers.
 */
export interface LatticePoint {
    readonly x: bigint;
    readonly y: bigint;
}

/**
 * Hexagonal compass directions on a pointy-top axial lattice (0 to 5).
 * 0: EAST (+1, 0)
 * 1: SOUTH_EAST (+1, -1)
 * 2: SOUTH_WEST (0, -1)
 * 3: WEST (-1, 0)
 * 4: NORTH_WEST (-1, +1)
 * 5: NORTH_EAST (0, +1)
 */
export type HexDirection = 0 | 1 | 2 | 3 | 4 | 5;

/**
 * Six exact integer vertices defining a pointy-top hexagon in doubled affine space.
 */
export type HexVertices = readonly [
    LatticePoint, // V0: (cx, cy + 4)     Top
    LatticePoint, // V1: (cx + 2, cy + 2) Top-Right
    LatticePoint, // V2: (cx + 2, cy - 2) Bottom-Right
    LatticePoint, // V3: (cx, cy - 4)     Bottom
    LatticePoint, // V4: (cx - 2, cy - 2) Bottom-Left

/**
 * ============================================================================
 * :: LATTICE INVARIANT GUARDS & BRANDING FACTORIES
 * ============================================================================
 * Enforces planar cubic integer constraints, coordinate boundaries, and type
 * assertion guards with zero floating-point arithmetic.
 */

import type {
    TacticalCellId,
    TacticalEdgeId,
    AxialCoord,
    LatticePoint,
} from './lattice.interface.js';

export const AXIAL_COORD_MIN = -32768;
export const AXIAL_COORD_MAX = 32767;

/**
 * Asserts that an input number is an exact safe integer within [-32768, 32767].
 */
export function assertIntegerCoordinate(value: number, paramName: string): void {
    if (!Number.isInteger(value)) {
        throw new TypeError(
            `INVALID_COORDINATE: Parameter '${paramName}' must be an exact integer. Received: ${value}`
        );
    }
    if (value < AXIAL_COORD_MIN || value > AXIAL_COORD_MAX) {
        throw new RangeError(
            `COORDINATE_OUT_OF_BOUNDS: Parameter '${paramName}' must be between ${AXIAL_COORD_MIN} and ${AXIAL_COORD_MAX}. Received: ${value}`
        );
    }
}

/**
 * Creates and validates an immutable AxialCoord, deriving s = -q - r.
 */
export function createAxialCoord(q: number, r: number): AxialCoord {
    assertIntegerCoordinate(q, 'q');
    assertIntegerCoordinate(r, 'r');
    const s = -q - r;
    assertIntegerCoordinate(s, 's');

    return Object.freeze({ q, r, s });
}

/**
 * Validates that an unknown object conforms strictly to the AxialCoord contract.
 */
export function isAxialCoord(val: unknown): val is AxialCoord {
    if (typeof val !== 'object' || val === null) {
        return false;
    }
    const candidate = val as Record<string, unknown>;
    const { q, r, s } = candidate;

    if (
        typeof q !== 'number' ||
        typeof r !== 'number' ||
        typeof s !== 'number'
    ) {
        return false;
    }

    if (!Number.isInteger(q) || !Number.isInteger(r) || !Number.isInteger(s)) {
        return false;
    }

    return q + r + s === 0;
}

/**
 * Asserts the planar cubic invariant q + r + s === 0.
 */
export function assertPlanarCubicInvariant(coord: AxialCoord): void {
    if (!Number.isInteger(coord.q) || !Number.isInteger(coord.r) || !Number.isInteger(coord.s)) {
        throw new TypeError('CUBIC_INVARIANT_BREACH: Coordinates must be exact integers.');
    }
    if (coord.q + coord.r + coord.s !== 0) {
        throw new Error(
            `CUBIC_INVARIANT_BREACH: q + r + s must equal 0. Received: q=${coord.q}, r=${coord.r}, s=${coord.s} (sum: ${coord.q + coord.r + coord.s})`
        );
    }
}

/**
 * Factory casting an unsigned 32-bit integer to a branded TacticalCellId.
 */
export function asTacticalCellId(packed: number): TacticalCellId {
    if (!Number.isInteger(packed) || packed < 0 || packed > 0xFFFFFFFF) {
        throw new TypeError(
            `INVALID_PACKED_CELL: TacticalCellId must be an unsigned 32-bit integer. Received: ${packed}`
        );
    }
    return packed as TacticalCellId;
}

/**
 * Factory casting an unsigned 64-bit BigInt to a branded TacticalEdgeId.
 */
export function asTacticalEdgeId(packed: bigint): TacticalEdgeId {
    if (typeof packed !== 'bigint' || packed < 0n || packed > 0xFFFFFFFFFFFFFFFFn) {
        throw new TypeError(
            `INVALID_PACKED_EDGE: TacticalEdgeId must be an unsigned 64-bit BigInt. Received: ${packed}`
        );
    }
    return packed as TacticalEdgeId;
}

/**
 * Factory for exact integer doubled affine LatticePoint.
 */
export function createLatticePoint(x: bigint | number, y: bigint | number): LatticePoint {
    const bx = typeof x === 'bigint' ? x : BigInt(Math.trunc(x));
    const by = typeof y === 'bigint' ? y : BigInt(Math.trunc(y));
    return Object.freeze({ x: bx, y: by });
}

        });

        it('rejects negative values for TacticalEdgeId', () => {
            expect(() => asTacticalEdgeId(-1n)).toThrow(TypeError);
        });
    });

    describe('Doubled Coordinate LatticePoint Construction', () => {
