```typescript
import { MenuActionPayload, SpatialBus } from '../fce/menu.interface.js';
import { createAxialCoord } from '../../28.axial.unit/val/axial.model.js';
import {
  createInProcessSpatialPort,
  createSpatialKernel,
  PureSpatialKernel,
  SpatialPort
} from '../../00.spatial.unit/buz/spatial.buzz.js';
import { PerceptCandidate } from '../../42.receipt.unit/fce/receipt.interface.js';
import { canonicalizeJCS, sha256Hex } from '../../19.ledger.unit/buz/ledger.buzz.js';
import { ScrubbedRaycastQuery } from '../../14.port.unit/fce/port.interface.js';

export function createReceiptVerificationAction(port?: SpatialPort): MenuActionPayload {
  const activePort = port ?? createInProcessSpatialPort(createSpatialKernel());

  return {
    id: 'REC01_INSPECT_RECEIPT',
    label: '[REC01] Inspect Query Hash Receipt',
    shortcut: 'REC01',
    category: 'DIAGNOSTIC',
    execute: async (bus: SpatialBus) => {
      const origin = createAxialCoord(0, 0);
      const target = createAxialCoord(2, 0);

      const rawQuery: ScrubbedRaycastQuery = {
        queryType: 'RAYCAST',
        origin,
        target,
        thresholdBps: 500,
        calculationMode: 'SUBTRACTIVE_LINEAR'
      };

      bus.emit('TELEMETRY_LOG', {
        message: ':: [REC01] Initiating Synchronous Query & Cryptographic Audit...'
      });

      const start = performance.now();
      const receipt = activePort.invokeSynchronous<PerceptCandidate>(rawQuery);
      const durationMs = performance.now() - start;

      if (!receipt.ok) {
        bus.emit('TELEMETRY_LOG', {
          message: `:: [REC01] Query Execution Failed: ${receipt.error ?? 'UNKNOWN_ERROR'}`
        });
        return;
      }

      const percept = receipt.result;
      const quad = receipt.auditQuadruple;

      // 1. Recalculate RFC 8785 Canonical JCS SHA-256 Hashes
      const computedQueryJcs = canonicalizeJCS(rawQuery);
      const computedReceiptJcs = canonicalizeJCS(percept);
      const computedQueryHash = sha256Hex(computedQueryJcs);
      const computedReceiptHash = sha256Hex(computedReceiptJcs);

      // 2. Parity Verifications
      const queryMatch = computedQueryHash === quad.queryHash;
      const receiptMatch = computedReceiptHash === quad.receiptHash;
      const allPassed = queryMatch && receiptMatch;

      // 3. Emit 7-Bit ASCII Telemetry Stream to cns00
      bus.emit('TELEMETRY_LOG', {
        message: `:: Query Type: ${receipt.queryType} | Origin: (0,0) -> Target: (2,0)`
      });
      bus.emit('TELEMETRY_LOG', {
        message: `:: Percept: Modality=${percept.channel} | Quality=${percept.qualityPermille}/1000 | Occluded=${percept.isOccluded}`
      });
      bus.emit('TELEMETRY_LOG', {
        message: `:: Topology Root Hash: ${quad.topologyHash.slice(0, 16)}...`
      });
      bus.emit('TELEMETRY_LOG', {
        message: `:: Overlay Root Hash:  ${quad.overlayHash.slice(0, 16)}...`
      });
      bus.emit('TELEMETRY_LOG', {
        message: `:: Query Hash:   ${quad.queryHash.slice(0, 16)}... [${queryMatch ? 'MATCH: OK' : 'MISMATCH: FAIL'}]`
      });
      bus.emit('TELEMETRY_LOG', {
        message: `:: Receipt Hash: ${quad.receiptHash.slice(0, 16)}... [${receiptMatch ? 'MATCH: OK' : 'MISMATCH: FAIL'}]`
      });
      bus.emit('TELEMETRY_LOG', {
        message: `:: Audit Verification: ${allPassed ? '[100% PARITY CERTIFIED: OK]' : '[CRYPTO DIVERGENCE: FAIL]'}`
      });
      bus.emit('TELEMETRY_LOG', {
        message: `:: Latency: ${durationMs.toFixed(3)} ms (Isolate Budget <= 2.0 ms)`
      });
    }
  };
}
```

```typescript
import { createSpatialBus } from '../99.bus.unit/buz/bus.buzz.js';
import { MenuDispatcher } from '../98.menu.unit/buz/00.menu.buzz.js';
import { createLatticeInspectionAction } from '../98.menu.unit/buz/01.lattice-menu.buzz.js';
import { createTopologyLoaderAction } from '../98.menu.unit/buz/02.topology-menu.buzz.js';
import { createTopologyDatabase } from '../07.database.unit/buz/database.buzz.js';
import { createLoSVisualizerAction } from '../98.menu.unit/buz/03.los-menu.buzz.js';
import { createSoundPropagationAction } from '../98.menu.unit/buz/04.sound-menu.buzz.js';
import { createPathSimulationAction } from '../98.menu.unit/buz/05.path-menu.buzz.js';
import { createDisplacementAction } from '../98.menu.unit/buz/06.displacement-menu.buzz.js';
import { createReceiptVerificationAction } from '../98.menu.unit/buz/07.receipt-menu.buzz.js';

export interface CockpitInstance {
  readonly bus: ReturnType<typeof createSpatialBus>;
  readonly menu: MenuDispatcher;
  readonly renderTelemetryFrame: () => string;
  readonly getActiveAsciiFrame: () => string | undefined;
  destroy(): void;
}

export function initializeCockpitHarness(): CockpitInstance {
  const bus = createSpatialBus();
  const menu = new MenuDispatcher(bus);
  const db = createTopologyDatabase();
  const logs: string[] = [];
  let lastAsciiFrame: string | undefined = undefined;

  // Register domain actions
  menu.registerAction(createLatticeInspectionAction());
  menu.registerAction(createTopologyLoaderAction(db));
  menu.registerAction(createLoSVisualizerAction());
  menu.registerAction(createSoundPropagationAction());
  menu.registerAction(createPathSimulationAction());
  menu.registerAction(createDisplacementAction());
  menu.registerAction(createReceiptVerificationAction());

  const unsubLogs = bus.subscribe('TELEMETRY_LOG', (evt) => {
    const msg = String(evt.payload.message || '');
    logs.push(msg);
    if (logs.length > 100) {
      logs.shift();
    }
  });

  const unsubFrame = bus.subscribe('ASCII_FRAME_RENDER', (evt) => {
    lastAsciiFrame = String(evt.payload.frame || '');
  });

  const renderTelemetryFrame = (): string => {
    const catalog = menu.getCatalog();
    const actionLines = catalog.actions
      .map((act, i) => {
        const marker = i === catalog.activeIndex ? '>' : ' ';
        return `  ${marker} [${act.shortcut}] ${act.label} (${act.category})`;
      })
      .join('\n');

    const recentLogs = logs.slice(-7).join('\n') || '  >> [SYSTEM READY: Awaiting Operator Input]';
    const frameSection = lastAsciiFrame ? `\n${lastAsciiFrame}\n` : '';

    return [
      '+--------------------------------------------------------------------------+',
      '| 000.goblin-spatial :: BLESSED TERMINAL COCKPIT (Phase 8)                |',
      '+--------------------------------------------------------------------------+',
      '| MENU CONTROLS (Columns 0-4, Rows 0-12)                                   |',
      actionLines,
      '+--------------------------------------------------------------------------+',
      '| TELEMETRY LOG [cns00] (Columns 4-12, Rows 3-12)                          |',
      recentLogs,
      frameSection,
      '+--------------------------------------------------------------------------+'
    ].join('\n');
  };

  return {
    bus,
    menu,
    renderTelemetryFrame,
    getActiveAsciiFrame: () => lastAsciiFrame,
    destroy: () => {
      unsubLogs();
      unsubFrame();
      bus.clear();
      db.clear();
    }
  };
}

if (process.env.NODE_ENV !== 'test' && typeof process.stdout.write === 'function') {
  const cockpit = initializeCockpitHarness();
  process.stdout.write(cockpit.renderTelemetryFrame() + '\n');
}

```
