function deepFreeze<T>(obj: T): Readonly<T> {
  if (obj === null || typeof obj !== 'object') {
    return obj;
  }
  Object.freeze(obj);
  for (const key of Object.keys(obj as object)) {
    const value = (obj as Record<string, unknown>)[key];
    if (value !== null && typeof value === 'object' && !Object.isFrozen(value)) {
      deepFreeze(value);
    }
  }
  return obj as Readonly<T>;
}

/**
 * RFC 8785 JSON Canonicalization Scheme (JCS) Serializer.
 * - Recursively sorts object keys lexicographically by UTF-16 code units.
 * - Preserves array sequence order strictly.
 * - Minifies whitespace outside strings.
 * - Rejects floating-point non-integers, NaN, and Infinity.
 */
export function canonicalizeJCS(value: unknown): string {
  if (value === null || typeof value !== 'object') {
    if (typeof value === 'number') {
      if (!Number.isFinite(value) || !Number.isInteger(value)) {
        throw new TypeError(`JCS_NON_INTEGER_OR_NON_FINITE: ${value}`);
      }
      return value.toString();
    }
    return JSON.stringify(value);
  }

  if (Array.isArray(value)) {
    return '[' + value.map((item) => canonicalizeJCS(item)).join(',') + ']';
  }

  const record = value as Record<string, unknown>;
  const sortedKeys = Object.keys(record).sort((a, b) => (a < b ? -1 : a > b ? 1 : 0));
  const elements = sortedKeys.map(
    (key) => `${JSON.stringify(key)}:${canonicalizeJCS(record[key])}`
  );
  return '{' + elements.join(',') + '}';
}

/**
 * Computes lowercase 64-char SHA-256 digest of arbitrary UTF-8 string.
 */
export function sha256Hex(content: string): string {
  return createHash('sha256').update(content, 'utf8').digest('hex');
}

/**
 * Computes deterministic genesis root hash for a given streamId:
 * H_0 = SHA-256("GOBLIN-SPATIAL-GENESIS::" || streamId)
 */
export function computeGenesisHash(streamId: string): string {
  return sha256Hex(`${GENESIS_PREFIX}${streamId}`);
}

/**
 * Mints recursive SHA-256 block hash for an entry:
 * H_n = SHA-256(H_{n-1} || "::" || seq || "::" || tick || "::" || streamId || "::" || queryType || "::" || CanonicalJCS(quadruple) || "::" || kernelVersion)
 */
export function mintEntryHash(
  previousHash: string,
  sequenceNumber: number,
  occurredAtTick: number,
  streamId: string,
  queryType: string,
  auditQuadruple: SpatialAuditQuadruple,
  kernelVersion: string
): string {
  const quadrupleJCS = canonicalizeJCS(auditQuadruple);
  const hashPayload = [
    previousHash,
    sequenceNumber.toString(),
    occurredAtTick.toString(),
    streamId,
    queryType,
    quadrupleJCS,
    kernelVersion
  ].join('::');

  return sha256Hex(hashPayload);
}

export class SpatialLedgerLogger {

  private readonly kernelVersion: string;
  private readonly genesisHash: string;

  }

   */
  public logTransaction(
    occurredAtTick: number,
    queryType: string,
    rawQuery: unknown,
    rawReceipt: unknown,
    contextHashes?: { topologyHash?: string; overlayHash?: string }
  ): SpatialLedgerEntry {
    this.sequenceCounter++;
    const sequenceNumber = this.sequenceCounter;

    const queryJCS = canonicalizeJCS(rawQuery);
    const receiptJCS = canonicalizeJCS(rawReceipt);

    const queryHash = sha256Hex(queryJCS);
    const receiptHash = sha256Hex(receiptJCS);

    const topologyHash =
      contextHashes?.topologyHash ?? sha256Hex('TOPOLOGY_DEFAULT_CANONICAL');
    const overlayHash =
      contextHashes?.overlayHash ?? sha256Hex('OVERLAY_EMPTY_IDENTITY');

    const auditQuadruple: SpatialAuditQuadruple = {
      topologyHash,
      overlayHash,
      queryHash,
      receiptHash
    };

    const previousEventHash = this.currentHeadHash;
    const eventHash = mintEntryHash(
      previousEventHash,
      sequenceNumber,
      occurredAtTick,
      this.streamId,
      queryType,
      auditQuadruple,
      this.kernelVersion
    );

    const entryCandidate: SpatialLedgerEntry = {
      sequenceNumber,
      occurredAtTick,
      streamId: this.streamId,
      queryType,
      auditQuadruple,
      kernelVersion: this.kernelVersion,
      previousEventHash,
      eventHash
    };

    const parsedEntry = SpatialLedgerEntrySchema.parse(entryCandidate);
    const sealedEntry = deepFreeze(parsedEntry) as SpatialLedgerEntry;

    this.entries.push(sealedEntry);
    this.currentHeadHash = eventHash;

    return sealedEntry;
  }

  /**
   * Audits the entire internal hash chain from genesis to head.
   */
  public verifyIntegrity(): HashChainVerificationResult {
    return verifyLedgerIntegrity(this.entries, this.streamId);
  }

/**
 * Verifies any arbitrary sequence of ledger entries against genesis and RFC 8785 chaining rules.
 */
export function verifyLedgerIntegrity(
  entries: readonly SpatialLedgerEntry[],
  streamId: string
): HashChainVerificationResult {
  const genesisHash = computeGenesisHash(streamId);
  let expectedPrevHash = genesisHash;

  for (let i = 0; i < entries.length; i++) {
    const entry = entries[i];
    const expectedSeq = i + 1;

    if (entry.sequenceNumber !== expectedSeq) {
      return {
        valid: false,
        totalEntries: entries.length,
        genesisHash,
        headHash: expectedPrevHash,
        brokenSequenceIndex: i,
        error: `SEQUENCE_GAP_DETECTED: Expected seq ${expectedSeq}, got ${entry.sequenceNumber}`
      };
    }

    if (entry.previousEventHash !== expectedPrevHash) {
      return {
        valid: false,
        totalEntries: entries.length,
        genesisHash,
        headHash: expectedPrevHash,
        brokenSequenceIndex: i,
        error: `PREVIOUS_HASH_MISMATCH: Entry ${entry.sequenceNumber} declared prev ${entry.previousEventHash}, expected ${expectedPrevHash}`
      };
    }

    const recalculatedHash = mintEntryHash(
      entry.previousEventHash,
      entry.sequenceNumber,
      entry.occurredAtTick,
      entry.streamId,
      entry.queryType,
      entry.auditQuadruple,
      entry.kernelVersion
    );

    if (entry.eventHash !== recalculatedHash) {
      return {
        valid: false,
        totalEntries: entries.length,
        genesisHash,
        headHash: expectedPrevHash,
        brokenSequenceIndex: i,
        error: `BLOCK_HASH_DIVERGENCE: Entry ${entry.sequenceNumber} hash ${entry.eventHash} != computed ${recalculatedHash}`
      };
    }

    expectedPrevHash = entry.eventHash;
  }

  return {
    valid: true,
    totalEntries: entries.length,
    genesisHash,
    headHash: expectedPrevHash
  };
}

    const unorderedObj = {
      z_coord: 0,
      a_name: 'test_node',
      m_metadata: {
        y_val: 200,
        b_val: 100
      }
    };

    const jcs = canonicalizeJCS(unorderedObj);
    // Keys must appear in strict alphabetical order: a_name, m_metadata (b_val, y_val), z_coord
    expect(jcs).toBe(
      '{"a_name":"test_node","m_metadata":{"b_val":100,"y_val":200},"z_coord":0}'
    );
  });

  it('rejects floating-point non-integers during canonicalization', () => {
    expect(() => canonicalizeJCS({ floatVal: 45.5 })).toThrow(TypeError);
    expect(() => canonicalizeJCS({ nanVal: NaN })).toThrow(TypeError);
    expect(() => canonicalizeJCS({ infVal: Infinity })).toThrow(TypeError);
  });

  it('initializes genesis hash rooted in the streamId', () => {
    const expectedGenesis = sha256Hex(`GOBLIN-SPATIAL-GENESIS::${streamId}`);
    expect(logger.getGenesisHash()).toBe(expectedGenesis);
    expect(logger.getHeadHash()).toBe(expectedGenesis);
    expect(logger.getEntryCount()).toBe(0);

  it('appends spatial transactions and mints recursive SHA-256 hash chains', () => {

    const receipt1 = { isVisible: true, transmissionBps: 10000 };

    expect(entry1.occurredAtTick).toBe(100);

    expect(entry2.sequenceNumber).toBe(2);
    expect(entry2.occurredAtTick).toBe(105);

    expect(logger.getHeadHash()).toBe(entry2.eventHash);
    expect(logger.getEntryCount()).toBe(2);

  it('verifies 100% hash chain integrity on un-tampered logs', () => {
    logger.logTransaction(1, 'QUERY_A', { a: 1 }, { res: 10 });
    logger.logTransaction(2, 'QUERY_B', { b: 2 }, { res: 20 });
    logger.logTransaction(3, 'QUERY_C', { c: 3 }, { res: 30 });

    const audit = logger.verifyIntegrity();

    expect(audit.totalEntries).toBe(3);
    expect(audit.headHash).toBe(logger.getHeadHash());
    expect(audit.error).toBeUndefined();

  it('detects tampering and hash divergence when an entry payload is mutated', () => {
    logger.logTransaction(1, 'STEP_1', { step: 1 }, { cost: 100 });
    logger.logTransaction(2, 'STEP_2', { step: 2 }, { cost: 200 });
    logger.logTransaction(3, 'STEP_3', { step: 3 }, { cost: 300 });

    const entries = [...logger.getEntries()] as SpatialLedgerEntry[];

    // Simulate tampering: retroactively mutate entry 2's queryHash
    const tamperedEntry: SpatialLedgerEntry = {
      ...entries[1],
      auditQuadruple: {
        ...entries[1].auditQuadruple,
        queryHash: sha256Hex('FORGED_QUERY_DATA')
      }
    };
    entries[1] = tamperedEntry;

    const audit = verifyLedgerIntegrity(entries, streamId);
    expect(audit.valid).toBe(false);
    expect(audit.brokenSequenceIndex).toBe(1);

    logger.logTransaction(2, 'STEP_2', { step: 2 }, { cost: 200 });
    logger.logTransaction(3, 'STEP_3', { step: 3 }, { cost: 300 });

    const entries = logger.getEntries();

    const gappedEntries = [entries[0], entries[2]];

    expect(audit.brokenSequenceIndex).toBe(1);

  });

  it('exports valid newline-delimited JSON (ndjson)', () => {
    logger.logTransaction(1, 'TEST_A', { q: 1 }, { r: 1 });
    logger.logTransaction(2, 'TEST_B', { q: 2 }, { r: 2 });

    expect(parsedFirst.sequenceNumber).toBe(1);
    expect(parsedFirst.eventHash).toBe(logger.getEntries()[0].eventHash);
  });

  it('enforces deep-freeze runtime immutability on returned entries', () => {
    const entry = logger.logTransaction(1, 'IMMUTABLE_TEST', { test: 1 }, { ok: true });

    expect(Object.isFrozen(entry)).toBe(true);
    expect(Object.isFrozen(entry.auditQuadruple)).toBe(true);

    expect(() => {
      // @ts-expect-error - testing immutability

    }).toThrow(TypeError);
