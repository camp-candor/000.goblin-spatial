```typescript
import { BasisPoints } from '../fce/witness.interface.js';

export const MAX_BASIS_POINTS = 10000;
export const MIN_BASIS_POINTS = 0;

export function toBasisPoints(value: number): BasisPoints {
  if (!Number.isInteger(value)) {
    throw new TypeError(`BASIS_POINTS_NON_INTEGER: ${value}`);
  }
  if (value < MIN_BASIS_POINTS || value > MAX_BASIS_POINTS) {
    throw new RangeError(`BASIS_POINTS_OUT_OF_BOUNDS: ${value} not in [0, 10000]`);
  }
  return value as BasisPoints;
}

export function clampBasisPoints(value: number): BasisPoints {
  if (!Number.isInteger(value)) {
    throw new TypeError(`CLAMP_NON_INTEGER: ${value}`);
  }
  if (value < MIN_BASIS_POINTS) return MIN_BASIS_POINTS as BasisPoints;
  if (value > MAX_BASIS_POINTS) return MAX_BASIS_POINTS as BasisPoints;
  return value as BasisPoints;
}

/**
 * Integer Round-Half-Even (Banker's Rounding) for Basis Point Scaling:
 * Scale(A, B) = round_half_even(A * B / 10000)
 */
export function scaleBasisPointsRoundHalfEven(a: BasisPoints, b: BasisPoints): BasisPoints {
  const product = BigInt(a) * BigInt(b);
  const denom = 10000n;
  const quotient = product / denom;
  const remainder = product % denom;
  const half = 5000n;

  let rounded = quotient;
  if (remainder > half) {
    rounded = quotient + 1n;
  } else if (remainder === half) {
    // Round to nearest even quotient
    if (quotient % 2n !== 0n) {
      rounded = quotient + 1n;
    }
  }

  return clampBasisPoints(Number(rounded));
}

```

### 3. `packages/301.spatial/40.witness.unit/buz/witness.buzz.ts`

```typescript
import { AxialCoord, packAxial, PackedAxial } from '../../28.axial.unit/val/axial.model.js';
import {
  BasisPoints,
  CellOpacitySample,
  OpticalTransmissionQuery,
  OpticalTransmissionReceipt
} from '../fce/witness.interface.js';
import {
  MAX_BASIS_POINTS,
  MIN_BASIS_POINTS,
  clampBasisPoints,
  toBasisPoints,
  scaleBasisPointsRoundHalfEven
} from '../val/witness.model.js';

export function computeOpticalTransmission(
  query: OpticalTransmissionQuery,
  getCellOpacity: (packed: PackedAxial) => BasisPoints
): OpticalTransmissionReceipt {
  const originPacked = packAxial(query.origin.q, query.origin.r);
  const targetPacked = packAxial(query.target.q, query.target.r);
  const threshold = query.thresholdBps !== undefined ? query.thresholdBps : toBasisPoints(0);
  const mode = query.calculationMode || 'SUBTRACTIVE_LINEAR';

  if (originPacked === targetPacked) {
    return {
      originPacked,
      targetPacked,
      pathLength: 0,
      accumulatedOpacityBps: toBasisPoints(0),
      transmissionBps: toBasisPoints(MAX_BASIS_POINTS),
      isVisible: true,
      samples: []
    };
  }

  const samples: CellOpacitySample[] = [];
  let totalOpacity = 0;
  let transmittanceBps = MAX_BASIS_POINTS;
  let severedAtCell: PackedAxial | undefined = undefined;

  for (let i = 0; i < query.traversedCells.length; i++) {
    const coord = query.traversedCells[i];
    const packed = packAxial(coord.q, coord.r);
    const isEndpoint = packed === originPacked || packed === targetPacked;

    // Endpoint Exemption: Endpoints count as completely transparent (0 bp opacity)
    const effectiveOpacity = isEndpoint ? toBasisPoints(0) : getCellOpacity(packed);

    samples.push({
      coord,
      packed,
      opacityBps: effectiveOpacity,
      isEndpoint
    });

    if (isEndpoint) {
      continue;
    }

    if (mode === 'SUBTRACTIVE_LINEAR') {
      totalOpacity += effectiveOpacity;
      if (totalOpacity >= MAX_BASIS_POINTS && severedAtCell === undefined) {
        severedAtCell = packed;
      }
    } else {
      // Multiplicative Beer-Lambert Formulation:
      // T_{k+1} = scaleBasisPoints(T_k, 10000 - opacity)
      const remainingTransparency = toBasisPoints(MAX_BASIS_POINTS - effectiveOpacity);
      transmittanceBps = scaleBasisPointsRoundHalfEven(
        toBasisPoints(transmittanceBps),
        remainingTransparency
      );
      if (transmittanceBps <= 0 && severedAtCell === undefined) {
        severedAtCell = packed;
      }
    }
  }

  let finalTransmission: BasisPoints;
  let finalAccumulatedOpacity: BasisPoints;

  if (mode === 'SUBTRACTIVE_LINEAR') {
    finalAccumulatedOpacity = clampBasisPoints(totalOpacity);
    finalTransmission = clampBasisPoints(MAX_BASIS_POINTS - finalAccumulatedOpacity);
  } else {
    finalTransmission = clampBasisPoints(transmittanceBps);
    finalAccumulatedOpacity = clampBasisPoints(MAX_BASIS_POINTS - finalTransmission);
  }

  const isVisible = finalTransmission > threshold;

  return {
    originPacked,
    targetPacked,
    pathLength: query.traversedCells.length,
    accumulatedOpacityBps: finalAccumulatedOpacity,
    transmissionBps: finalTransmission,
    isVisible,
    samples,
    severedAtCell
  };
}
```
