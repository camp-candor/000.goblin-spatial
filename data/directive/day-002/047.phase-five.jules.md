    const dq = Math.abs(query.target.q - query.origin.q);
    const dr = Math.abs(query.target.r - query.origin.r);
    const topologicalDistance = Math.max(dq, dr, Math.abs(query.target.s - query.origin.s));

    // Linear integer attenuation: 10,000 bps minus 500 bps per hex distance

    const transmissionBps = Math.max(0, Math.min(10000, rawTransmission));
    const isOccluded = transmissionBps < 5000;

    return Object.freeze({
      ok: true,
      transmissionBps,
      topologicalDistance,
      isOccluded,
      occludingCoord: isOccluded ? query.target : null,
    });

    const dr = Math.abs(query.destination.r - query.origin.r);
    const ds = Math.abs(query.destination.s - query.origin.s);
    const hexDist = Math.max(dq, dr, ds);

    const cellsExpanded = hexDist * 16;
    if (cellsExpanded >= query.workEnvelopeLimit) {
      return Object.freeze({
        ok: false,
        path: Object.freeze([]),
        totalCostUnits: 0,
        cellsExpanded,
        failureCode: 'BUDGET_EXCEEDED',
      });
    }

    const startPacked = packAxialCell(query.origin.q, query.origin.r);
    const destPacked = packAxialCell(query.destination.q, query.destination.r);

        ok: false,
        path: Object.freeze([]),

        cellsExpanded,
        failureCode: 'UNREACHABLE',
      });

      ok: true,
      path: Object.freeze([startPacked, destPacked]),
      totalCostUnits,
      cellsExpanded,
    });
  }

    const originPacked = packAxialCell(query.origin.q, query.origin.r);

import type {
  SpatialPort,
  SpatialExecutionContext,
  SynchronousExecutionReceipt,
  ScrubbedSpatialQuery,
} from '../../14.port.unit/fce/port.interface.js';
import type {
  PureSpatialKernel,
} from '../fce/spatial.interface.js';
import { SpatialKernelInvoker } from './spatial.buzz.js';
import { sanitizeSpatialQueryPayload } from '../../14.port.unit/buz/port.buzz.js';
import { canonicalizeJson, computeSha256Hex } from '../../19.ledger.unit/buz/ledger.buzz.js';
import { deepFreezeReceipt } from '../../42.receipt.unit/buz/receipt.buzz.js';
import { GENESIS_HASH_64 } from '../../19.ledger.unit/val/ledger.model.js';
import { packAxialCell } from '../../38.types.unit/fce/manifest.schema.js';

export class InProcessSpatialAdapter implements SpatialPort {
  readonly kernelVersion: string;
  private readonly kernel: PureSpatialKernel;

  constructor(kernel?: PureSpatialKernel) {
    this.kernel = kernel ?? new SpatialKernelInvoker();
    this.kernelVersion = this.kernel.kernelVersion;
  }

  invokeSynchronous<T = unknown>(
    rawQuery: unknown,
    context?: SpatialExecutionContext
  ): SynchronousExecutionReceipt<T> {
    const startHr = typeof performance !== 'undefined' ? performance.now() : Date.now();

    // 1. Epistemic Firewall: sanitize raw query and purge narrative properties
    const sanitizeResult = sanitizeSpatialQueryPayload(rawQuery);
    if (!sanitizeResult.valid || !sanitizeResult.scrubbedQuery) {
      const elapsedMicros = Math.round(
        ((typeof performance !== 'undefined' ? performance.now() : Date.now()) - startHr) * 1000
      );
      return deepFreezeReceipt({
        ok: false,
        queryType: (rawQuery as any)?.queryType ?? 'RAYCAST',
        kernelVersion: this.kernelVersion,
        payload: null as unknown as T,
        cellsExpanded: 0,
        failureCode: 'SCHEMA_BREACH',
        executionMicros: elapsedMicros,
      });
    }

    const query: ScrubbedSpatialQuery = sanitizeResult.scrubbedQuery;
    let ok = false;
    let payloadData: unknown = null;
    let cellsExpanded = 0;
    let failureCode: SynchronousExecutionReceipt['failureCode'];

    // 2. Dispatch query directly to synchronous mathematical kernel
    switch (query.queryType) {
      case 'RAYCAST': {
        const res = this.kernel.executeRaycast(query);
        ok = res.ok;
        cellsExpanded = res.topologicalDistance;
        payloadData = {
          sourceCoord: query.origin,
          observerCoord: query.target,
          qualityScorePerMille: Math.round(res.transmissionBps / 10),
          arrivalBearing: 0,
          isOccluded: res.isOccluded,
          evidence: {
            modality: 'OPTICAL',
            transmissionBps: res.transmissionBps,
            topologicalDistance: res.topologicalDistance,
            occludingBarrierCoord: res.occludingCoord,
          },
        };
        break;
      }

      case 'BOUNDED_PATH': {
        const res = this.kernel.executeBoundedPath(query);
        ok = res.ok;
        cellsExpanded = res.cellsExpanded;
        failureCode = res.failureCode;
        payloadData = res.ok
          ? {
              ok: true,
              originPacked: packAxialCell(query.origin.q, query.origin.r),
              destinationPacked: packAxialCell(query.destination.q, query.destination.r),
              path: res.path,
              totalCostUnits: res.totalCostUnits,
              stepsExpanded: res.cellsExpanded,
            }
          : {
              ok: false,
              originPacked: packAxialCell(query.origin.q, query.origin.r),
              destinationPacked: packAxialCell(query.destination.q, query.destination.r),
              failureCode: res.failureCode,
              stepsExpanded: res.cellsExpanded,
            };
        break;
      }

      case 'REACHABILITY': {
        const res = this.kernel.executeReachability(query);
        ok = true;
        cellsExpanded = res.length;
        payloadData = { reachableCells: res };
        break;
      }

      case 'DISPLACEMENT': {
        const res = this.kernel.executeDisplacement(query);
        ok = res.ok;
        failureCode = res.failureCode;
        payloadData = {
          ok: res.ok,
          targetCellPacked: res.targetCellPacked,
          resolvedCellPacked: res.resolvedCellPacked,
          relocationCostUnits: res.relocationCostUnits,
          wasEvicted: res.wasEvicted,
        };
        break;
      }
    }

    // 3. Calculate deterministic RFC 8785 hashes and quadruple
    const queryHash = computeSha256Hex(canonicalizeJson(query));
    const receiptHash = computeSha256Hex(canonicalizeJson(payloadData));
    const topologyHash = context?.topologyHash ?? GENESIS_HASH_64;
    const overlayHash = context?.overlayHash ?? GENESIS_HASH_64;

    const auditQuadruple = Object.freeze({
      topologyHash,
      overlayHash,
      queryHash,
      receiptHash,
    });

    // 4. Bind audit quadruple to sealed payload
    if (payloadData && typeof payloadData === 'object') {
      (payloadData as any).auditQuadruple = auditQuadruple;
    }

    const elapsedMicros = Math.round(
      ((typeof performance !== 'undefined' ? performance.now() : Date.now()) - startHr) * 1000
    );

    const receipt: SynchronousExecutionReceipt<T> = {
      ok,
      queryType: query.queryType,
      kernelVersion: this.kernelVersion,
      payload: payloadData as T,
      cellsExpanded,
      failureCode,
      executionMicros: elapsedMicros,
    };

    return deepFreezeReceipt(receipt);
  }
}

export function createInProcessSpatialPort(kernel?: PureSpatialKernel): SpatialPort {
  return new InProcessSpatialAdapter(kernel);
}

import clone from 'clone-deep';
import * as Act from './spatial.action.js';
import { SpatialModel } from './spatial.model.js';
import * as Buzz from './spatial.buzzer.js';
import type State from '../99.core/state.js';

export function reducer(
  model: SpatialModel = new SpatialModel(),
  act: Act.Actions,
  state?: State
) {
  switch (act.type) {
    case Act.INIT_SPATIAL:
      return Buzz.initSpatial(clone(model), act.bale, state);

    case Act.INVOKE_SYNCHRONOUS: {
      const port = Buzz.createInProcessSpatialPort();
      const rawQuery = act.bale?.dat?.rawQuery;
      const context = act.bale?.dat?.context;
      const receipt = port.invokeSynchronous(rawQuery, context);

      const next = clone(model);
      next.totalSynchronousInvocations += 1;
      next.lastExecutionMicros = receipt.executionMicros;
      if (receipt.failureCode === 'BUDGET_EXCEEDED') {
        next.totalBudgetsTripped += 1;
      }

      if (act.bale?.slv) {
        act.bale.slv({
          sptBit: {
            idx: receipt.ok ? 'invoke-success' : 'invoke-failure',
            val: receipt.ok ? 1 : 0,
            dat: { receipt },
          },
        });
      }
      return next;
    }

    default:
      return model;
  }
}
