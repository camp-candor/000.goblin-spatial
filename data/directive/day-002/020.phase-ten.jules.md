import { AxialCoord, PackedAxial, unpackAxial } from '../../28.axial.unit/val/axial.model.js';
import { chordDistanceSquared } from '../../28.axial.unit/buz/axial.buzz.js';

export interface TieBreakNode {
  readonly cell: PackedAxial;
  readonly fScore: number;
  readonly gScore: number;
  readonly hScore: number;
  readonly seedDigest: number;
}

/**
 * 4-Tier Deterministic Pathfinding Comparator:
 * 1. Total estimated cost (fScore ascending)
 * 2. Exact integer chord distance squared in doubled coordinates (hScore ascending)
 * 3. Seed digest integer tie-breaker
 * 4. Packed Cell ID (numeric ascending)
 */
export function compareTieBreakNodes(
  a: TieBreakNode,
  b: TieBreakNode,
  targetCoord?: AxialCoord
): number {
  if (a.fScore !== b.fScore) {
    return a.fScore - b.fScore;
  }

  if (targetCoord !== undefined) {
    const coordA = unpackAxial(a.cell);
    const coordB = unpackAxial(b.cell);
    const chordSqA = chordDistanceSquared(coordA, targetCoord);
    const chordSqB = chordDistanceSquared(coordB, targetCoord);
    if (chordSqA !== chordSqB) {
      return chordSqA < chordSqB ? -1 : 1;
    }
  } else if (a.hScore !== b.hScore) {
    return a.hScore - b.hScore;
  }

  if (a.seedDigest !== b.seedDigest) {
    return a.seedDigest - b.seedDigest;
  }

  return a.cell - b.cell;
}

import {
  AxialCoord,
  PackedAxial,
  packAxial,
  unpackAxial
} from '../../28.axial.unit/val/axial.model.js';
import { AXIAL_DIRECTIONS } from '../../46.neighbor.unit/buz/neighbor.buzz.js';
import { DynamicOverlay } from '../../38.types.unit/fce/manifest.schema.js';

export interface ReachableCellsQuery {
  readonly origin: AxialCoord;
  readonly budgetCostUnits: number;
  readonly overlay?: DynamicOverlay;
  readonly blockedCells?: ReadonlySet<PackedAxial>;
  readonly workEnvelopeLimit?: number;
}

export interface ReachableCellsResult {
  readonly reachable: ReadonlyMap<PackedAxial, number>; // packed -> costUnits
  readonly expandedSteps: number;
  readonly breachedEnvelope: boolean;
}

export function computeReachableCells(query: ReachableCellsQuery): ReachableCellsResult {
  const envelopeLimit = query.workEnvelopeLimit ?? 512;
  const originPacked = packAxial(query.origin.q, query.origin.r);
  const blocked = query.blockedCells ?? new Set<PackedAxial>();

  const costs = new Map<PackedAxial, number>();
  costs.set(originPacked, 0);

  const queue: { packed: PackedAxial; cost: number }[] = [
    { packed: originPacked, cost: 0 }
  ];

  let steps = 0;
  let breached = false;

  while (queue.length > 0) {
    steps++;
    if (steps > envelopeLimit) {
      breached = true;
      break;
    }

    queue.sort((a, b) => a.cost - b.cost || a.packed - b.packed);
    const curr = queue.shift()!;

    const coord = unpackAxial(curr.packed);

    for (let i = 0; i < 6; i++) {
      const dir = AXIAL_DIRECTIONS[i];
      const neighborQ = coord.q + dir.q;
      const neighborR = coord.r + dir.r;
      const neighborPacked = packAxial(neighborQ, neighborR);

      if (blocked.has(neighborPacked)) {
        continue;
      }

      const stepCost = 100; // 100 CostUnits standard step
      const totalCost = curr.cost + stepCost;

      if (totalCost > query.budgetCostUnits) {
        continue;
      }

      const prevCost = costs.get(neighborPacked);
      if (prevCost === undefined || totalCost < prevCost) {
        costs.set(neighborPacked, totalCost);
        queue.push({ packed: neighborPacked, cost: totalCost });
      }
    }
  }

  return {
    reachable: costs,
    expandedSteps: steps,
    breachedEnvelope: breached
  };
}

import { AXIAL_DIRECTIONS } from '../../46.neighbor.unit/buz/neighbor.buzz.js';
import { DynamicOverlay } from '../../38.types.unit/fce/manifest.schema.js';
import { compareTieBreakNodes, TieBreakNode } from '../../25.distance.unit/buz/distance.buzz.js';

export interface PathQueryOptions {
  readonly origin: AxialCoord;
  readonly destination: AxialCoord;
  readonly maxBudgetCostUnits: number;
  readonly overlay?: DynamicOverlay;
  readonly blockedCells?: ReadonlySet<PackedAxial>;
  readonly workEnvelopeLimit?: number;
  readonly episodeSeed?: number;
}

export type PathQueryResult =
  | {
      readonly ok: true;
      readonly originPacked: PackedAxial;
      readonly destinationPacked: PackedAxial;
      readonly path: readonly AxialCoord[];
      readonly pathPacked: readonly PackedAxial[];
      readonly totalCostUnits: number;
      readonly expandedSteps: number;
    }
  | {
      readonly ok: false;
      readonly reason: 'UNREACHABLE' | 'BUDGET_EXCEEDED' | 'IMPASSABLE';
      readonly expandedSteps: number;
    };

export function findShortestBoundedPath(query: PathQueryOptions): PathQueryResult {
  const envelopeLimit = query.workEnvelopeLimit ?? 512;
  const originPacked = packAxial(query.origin.q, query.origin.r);
  const destPacked = packAxial(query.destination.q, query.destination.r);
  const blocked = query.blockedCells ?? new Set<PackedAxial>();

  if (blocked.has(originPacked) || blocked.has(destPacked)) {
    return { ok: false, reason: 'IMPASSABLE', expandedSteps: 0 };
  }

  if (originPacked === destPacked) {
    return {
      ok: true,
      originPacked,
      destinationPacked: destPacked,
      path: [query.origin],
      pathPacked: [originPacked],
      totalCostUnits: 0,
      expandedSteps: 0
    };
  }

  const gScore = new Map<PackedAxial, number>();
  const parent = new Map<PackedAxial, PackedAxial>();
  const queue: TieBreakNode[] = [];

  gScore.set(originPacked, 0);
  const initialH = axialDistance(query.origin, query.destination) * 100;
  queue.push({
    cell: originPacked,
    fScore: initialH,
    gScore: 0,
    hScore: initialH,
    seedDigest: (query.episodeSeed ?? 0) ^ originPacked
  });

  let steps = 0;

  while (queue.length > 0) {
    steps++;
    if (steps > envelopeLimit) {
      return { ok: false, reason: 'BUDGET_EXCEEDED', expandedSteps: steps };
    }

    queue.sort((a, b) => compareTieBreakNodes(a, b, query.destination));
    const current = queue.shift()!;

    if (current.cell === destPacked) {
      const pathPacked: PackedAxial[] = [];
      let curr: PackedAxial | undefined = destPacked;
      while (curr !== undefined) {
        pathPacked.unshift(curr);
        curr = parent.get(curr);
      }

      const pathCoords = pathPacked.map((p) => unpackAxial(p));
      return {
        ok: true,
        originPacked,
        destinationPacked: destPacked,
        path: pathCoords,
        pathPacked,
        totalCostUnits: current.gScore,
        expandedSteps: steps
      };
    }

    const currentCoord = unpackAxial(current.cell);

    for (let i = 0; i < 6; i++) {
      const dir = AXIAL_DIRECTIONS[i];
      const neighborQ = currentCoord.q + dir.q;
      const neighborR = currentCoord.r + dir.r;
      const neighborPacked = packAxial(neighborQ, neighborR);

      if (blocked.has(neighborPacked)) {
        continue;
      }

      const stepCost = 100;
      const tentativeG = current.gScore + stepCost;

      if (tentativeG > query.maxBudgetCostUnits) {
        continue;
      }

      const prevG = gScore.get(neighborPacked);
      if (prevG === undefined || tentativeG < prevG) {
        gScore.set(neighborPacked, tentativeG);
        parent.set(neighborPacked, current.cell);

        const neighborCoord = unpackAxial(neighborPacked);
        const h = axialDistance(neighborCoord, query.destination) * 100;
        queue.push({
          cell: neighborPacked,
          fScore: tentativeG + h,
          gScore: tentativeG,
          hScore: h,
          seedDigest: (query.episodeSeed ?? 0) ^ neighborPacked
        });
      }
    }
  }

  return { ok: false, reason: 'UNREACHABLE', expandedSteps: steps };
}

      budgetCostUnits,
      overlay: options?.overlay,
      blockedCells: options?.blockedCells,
      workEnvelopeLimit: options?.workEnvelopeLimit ?? 512
    };
    return computeReachableCells(query);
  }
}

    menu,
