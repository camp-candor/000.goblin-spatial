```typescript
export interface CellCapacitySpec {
  readonly cellCapacityDefault: number;
  readonly overrides?: ReadonlyMap<PackedAxial, number>;
}

export interface CapacityCheckReceipt {
  readonly cell: PackedAxial;
  readonly currentOccupancy: number;
  readonly maxCapacity: number;
  readonly isOvercrowded: boolean;
  readonly excessOccupants: number;
}

export class CellCapacityEvaluator {
  private readonly defaultCapacity: number;
  private readonly overrides: ReadonlyMap<PackedAxial, number>;

  constructor(spec?: CellCapacitySpec) {
    this.defaultCapacity = spec?.cellCapacityDefault ?? 2; // Default 2 adult goblins per cell
    this.overrides = spec?.overrides ?? new Map<PackedAxial, number>();
  }

  public getCapacity(cell: PackedAxial): number {
    return this.overrides.get(cell) ?? this.defaultCapacity;
  }

  public checkSaturation(cell: PackedAxial, currentOccupancy: number): CapacityCheckReceipt {
    if (!Number.isInteger(currentOccupancy) || currentOccupancy < 0) {
      throw new TypeError(`INVALID_OCCUPANCY_COUNT: ${currentOccupancy}`);
    }

    const maxCapacity = this.getCapacity(cell);
    const isOvercrowded = currentOccupancy > maxCapacity;
    const excessOccupants = isOvercrowded ? currentOccupancy - maxCapacity : 0;

    return {
      cell,
      currentOccupancy,
      maxCapacity,
      isOvercrowded,
      excessOccupants
    };
  }
}

export const createCapacityEvaluator = (spec?: CellCapacitySpec): CellCapacityEvaluator =>
  new CellCapacityEvaluator(spec);
```

```typescript
export class CellOccupancyTracker {
  private readonly counts = new Map<PackedAxial, number>();

  public getOccupancy(cell: PackedAxial): number {
    return this.counts.get(cell) ?? 0;
  }

  public increment(cell: PackedAxial): number {
    const next = (this.counts.get(cell) ?? 0) + 1;
    this.counts.set(cell, next);
    return next;
  }

  public decrement(cell: PackedAxial): number {
    const current = this.counts.get(cell) ?? 0;
    if (current <= 0) {
      throw new RangeError(`OCCUPANCY_UNDERFLOW: Attempted to decrement empty cell ${cell}`);
    }
    const next = current - 1;
    if (next === 0) {
      this.counts.delete(cell);
    } else {
      this.counts.set(cell, next);
    }
    return next;
  }

  public setOccupancy(cell: PackedAxial, count: number): void {
    if (!Number.isInteger(count) || count < 0) {
      throw new TypeError(`INVALID_OCCUPANCY_VALUE: ${count}`);
    }
    if (count === 0) {
      this.counts.delete(cell);
    } else {
      this.counts.set(cell, count);
    }
  }

  public snapshot(): ReadonlyMap<PackedAxial, number> {
    return new Map(this.counts);
  }

  public clear(): void {
    this.counts.clear();
  }
}

export const createOccupancyTracker = (): CellOccupancyTracker => new CellOccupancyTracker();
```

```typescript
import {
  AxialCoord,
  PackedAxial,
  packAxial,
  unpackAxial
} from '../../28.axial.unit/val/axial.model.js';
import { chordDistanceSquared } from '../../28.axial.unit/buz/axial.buzz.js';
import { AXIAL_DIRECTIONS } from '../../46.neighbor.unit/buz/neighbor.buzz.js';
import { CellCapacityEvaluator } from '../../51.capacity.unit/buz/capacity.buzz.js';
import { CellOccupancyTracker } from '../../44.occupancy.unit/buz/occupancy.buzz.js';

export interface DisplacementQuery {
  readonly congestedCell: AxialCoord;
  readonly budgetCostUnits?: number;
  readonly blockedCells?: ReadonlySet<PackedAxial>;
  readonly workEnvelopeLimit?: number;
  readonly seedDigest?: number;
}

export type DisplacementResult =
  | {
      readonly ok: true;
      readonly targetCell: AxialCoord;
      readonly targetPacked: PackedAxial;
      readonly traversalPath: readonly AxialCoord[];
      readonly traversalCostUnits: number;
      readonly expandedSteps: number;
    }
  | {
      readonly ok: false;
      readonly reason: 'NO_LEGAL_CELL' | 'BUDGET_EXCEEDED';
      readonly expandedSteps: number;
    };

export function resolveDisplacement(
  query: DisplacementQuery,
  capacityEvaluator: CellCapacityEvaluator,
  occupancyTracker: CellOccupancyTracker
): DisplacementResult {
  const originPacked = packAxial(query.congestedCell.q, query.congestedCell.r);
  const maxBudget = query.budgetCostUnits ?? 1000;
  const envelopeLimit = query.workEnvelopeLimit ?? 512;
  const blocked = query.blockedCells ?? new Set<PackedAxial>();
  const seed = query.seedDigest ?? 0;

  // Origin cell occupancy check excluding the displaced actor (-1)
  const originOccupancy = Math.max(0, occupancyTracker.getOccupancy(originPacked) - 1);
  if (originOccupancy < capacityEvaluator.getCapacity(originPacked)) {
    // If excluding the evicted actor brings the cell back within legal capacity, no move is needed
    return {
      ok: true,
      targetCell: query.congestedCell,
      targetPacked: originPacked,
      traversalPath: [query.congestedCell],
      traversalCostUnits: 0,
      expandedSteps: 0
    };
  }

  type DijkstraNode = {
    readonly packed: PackedAxial;
    readonly cost: number;
    readonly chordSq: bigint;
    readonly tieDigest: number;
  };

  const gScore = new Map<PackedAxial, number>();
  const parent = new Map<PackedAxial, PackedAxial>();
  const queue: DijkstraNode[] = [];

  gScore.set(originPacked, 0);
  queue.push({
    packed: originPacked,
    cost: 0,
    chordSq: 0n,
    tieDigest: seed ^ originPacked
  });

  let steps = 0;

  while (queue.length > 0) {
    steps++;
    if (steps > envelopeLimit) {
      return { ok: false, reason: 'BUDGET_EXCEEDED', expandedSteps: steps };
    }

    // 4-Tier Deterministic Tie-Breaking Comparator
    queue.sort((a, b) => {
      if (a.cost !== b.cost) return a.cost - b.cost;
      if (a.chordSq !== b.chordSq) return a.chordSq < b.chordSq ? -1 : 1;
      if (a.tieDigest !== b.tieDigest) return a.tieDigest - b.tieDigest;
      return a.packed - b.packed;
    });

    const current = queue.shift()!;

    // Check capacity: Destination must NOT be the congested origin and must have remaining space
    if (current.packed !== originPacked) {
      const cellOcc = occupancyTracker.getOccupancy(current.packed);
      const cellCap = capacityEvaluator.getCapacity(current.packed);

      if (cellOcc < cellCap) {
        // Legal capacity cell found: reconstruct path
        const pathPacked: PackedAxial[] = [];
        let curr: PackedAxial | undefined = current.packed;
        while (curr !== undefined) {
          pathPacked.unshift(curr);
          curr = parent.get(curr);
        }

        const traversalPath = pathPacked.map((p) => unpackAxial(p));
        return {
          ok: true,
          targetCell: unpackAxial(current.packed),
          targetPacked: current.packed,
          traversalPath,
          traversalCostUnits: current.cost,
          expandedSteps: steps
        };
      }
    }

    const currentCoord = unpackAxial(current.packed);

    for (let i = 0; i < 6; i++) {
      const dir = AXIAL_DIRECTIONS[i];
      const neighborQ = currentCoord.q + dir.q;
      const neighborR = currentCoord.r + dir.r;
      const neighborPacked = packAxial(neighborQ, neighborR);

      if (blocked.has(neighborPacked)) {
        continue;
      }

      const stepCost = 100; // Standard integer 100 CostUnits per hex
      const tentativeG = current.cost + stepCost;

      if (tentativeG > maxBudget) {
        continue;
      }

      const prevG = gScore.get(neighborPacked);
      if (prevG === undefined || tentativeG < prevG) {
        gScore.set(neighborPacked, tentativeG);
        parent.set(neighborPacked, current.packed);

        const neighborCoord = unpackAxial(neighborPacked);
        const chordSq = chordDistanceSquared(neighborCoord, query.congestedCell);

        queue.push({
          packed: neighborPacked,
          cost: tentativeG,
          chordSq,
          tieDigest: seed ^ neighborPacked
        });
      }
    }
  }

  return { ok: false, reason: 'NO_LEGAL_CELL', expandedSteps: steps };
}
```

```typescript
import { createSpatialBus } from '../99.bus.unit/buz/bus.buzz.js';
import { MenuDispatcher } from '../98.menu.unit/buz/00.menu.buzz.js';
import { createLatticeInspectionAction } from '../98.menu.unit/buz/01.lattice-menu.buzz.js';
import { createTopologyLoaderAction } from '../98.menu.unit/buz/02.topology-menu.buzz.js';
import { createTopologyDatabase } from '../07.database.unit/buz/database.buzz.js';
import { createLoSVisualizerAction } from '../98.menu.unit/buz/03.los-menu.buzz.js';
import { createSoundPropagationAction } from '../98.menu.unit/buz/04.sound-menu.buzz.js';
import { createPathSimulationAction } from '../98.menu.unit/buz/05.path-menu.buzz.js';
import { createDisplacementAction } from '../98.menu.unit/buz/06.displacement-menu.buzz.js';

export interface CockpitInstance {
  readonly bus: ReturnType<typeof createSpatialBus>;
  readonly menu: MenuDispatcher;
  readonly renderTelemetryFrame: () => string;
  readonly getActiveAsciiFrame: () => string | undefined;
  destroy(): void;
}

export function initializeCockpitHarness(): CockpitInstance {
  const bus = createSpatialBus();
  const menu = new MenuDispatcher(bus);
  const db = createTopologyDatabase();
  const logs: string[] = [];
  let lastAsciiFrame: string | undefined = undefined;

  // Register domain actions
  menu.registerAction(createLatticeInspectionAction());
  menu.registerAction(createTopologyLoaderAction(db));
  menu.registerAction(createLoSVisualizerAction());
  menu.registerAction(createSoundPropagationAction());
  menu.registerAction(createPathSimulationAction());
  menu.registerAction(createDisplacementAction());

  const unsubLogs = bus.subscribe('TELEMETRY_LOG', (evt) => {
    const msg = String(evt.payload.message || '');
    logs.push(msg);
    if (logs.length > 100) {
      logs.shift();
    }
  });

  const unsubFrame = bus.subscribe('ASCII_FRAME_RENDER', (evt) => {
    lastAsciiFrame = String(evt.payload.frame || '');
  });

  const renderTelemetryFrame = (): string => {
    const catalog = menu.getCatalog();
    const actionLines = catalog.actions
      .map((act, i) => {
        const marker = i === catalog.activeIndex ? '>' : ' ';
        return `  ${marker} [${act.shortcut}] ${act.label} (${act.category})`;
      })
      .join('\n');

    const recentLogs = logs.slice(-6).join('\n') || '  >> [SYSTEM READY: Awaiting Operator Input]';
    const frameSection = lastAsciiFrame ? `\n${lastAsciiFrame}\n` : '';

    return [
      '+--------------------------------------------------------------------------+',
      '| 000.goblin-spatial :: BLESSED TERMINAL COCKPIT (Phase 7)                |',
      '+--------------------------------------------------------------------------+',
      '| MENU CONTROLS (Columns 0-4, Rows 0-12)                                   |',
      actionLines,
      '+--------------------------------------------------------------------------+',
      '| TELEMETRY LOG [cns00] (Columns 4-12, Rows 3-12)                          |',
      recentLogs,
      frameSection,
      '+--------------------------------------------------------------------------+'
    ].join('\n');
  };

  return {
    bus,
    menu,
    renderTelemetryFrame,
    getActiveAsciiFrame: () => lastAsciiFrame,
    destroy: () => {
      unsubLogs();
      unsubFrame();
      bus.clear();
      db.clear();
    }
  };
}

if (process.env.NODE_ENV !== 'test' && typeof process.stdout.write === 'function') {
  const cockpit = initializeCockpitHarness();
  process.stdout.write(cockpit.renderTelemetryFrame() + '\n');
}
```

### 6. `packages/301.spatial/test/displacement.test.ts`

```typescript
import { describe, it, expect, beforeEach } from 'vitest';
import { createAxialCoord, packAxial, PackedAxial } from '../28.axial.unit/val/axial.model.js';
import { createCapacityEvaluator, CellCapacityEvaluator } from '../51.capacity.unit/buz/capacity.buzz.js';
import { createOccupancyTracker, CellOccupancyTracker } from '../44.occupancy.unit/buz/occupancy.buzz.js';
import { resolveDisplacement } from '../50.displacement.unit/buz/displacement.buzz.js';
import { createDisplacementAction } from '../98.menu.unit/buz/06.displacement-menu.buzz.js';
import { createSpatialBus } from '../99.bus.unit/buz/bus.buzz.js';

describe('Phase 7 Spatial Capacity Displacement Eviction Engine', () => {
  let capacityEvaluator: CellCapacityEvaluator;
  let occupancyTracker: CellOccupancyTracker;

  beforeEach(() => {
    capacityEvaluator = createCapacityEvaluator({ cellCapacityDefault: 2 });
    occupancyTracker = createOccupancyTracker();
  });

  it('evaluates cell density saturation and flags overcrowding violations', () => {
    const cell = packAxial(0, 0);
    expect(capacityEvaluator.getCapacity(cell)).toBe(2);

    const checkNominal = capacityEvaluator.checkSaturation(cell, 2);
    expect(checkNominal.isOvercrowded).toBe(false);
    expect(checkNominal.excessOccupants).toBe(0);

    const checkOvercrowded = capacityEvaluator.checkSaturation(cell, 3);
    expect(checkOvercrowded.isOvercrowded).toBe(true);
    expect(checkOvercrowded.excessOccupants).toBe(1);
  });

  it('tracks atomic cell occupancy increments and decrements cleanly', () => {
    const cell = packAxial(1, -1);
    expect(occupancyTracker.getOccupancy(cell)).toBe(0);

    expect(occupancyTracker.increment(cell)).toBe(1);
    expect(occupancyTracker.increment(cell)).toBe(2);
    expect(occupancyTracker.decrement(cell)).toBe(1);
    expect(occupancyTracker.decrement(cell)).toBe(0);
    expect(() => occupancyTracker.decrement(cell)).toThrow(RangeError);
  });

  it('exempts the displaced actor so legal occupancy requires no movement', () => {
    const origin = createAxialCoord(0, 0);
    const originPacked = packAxial(0, 0);

    // Cell capacity 2, currently has 2 occupants. If one departs (2 - 1 = 1), it is legal!
    occupancyTracker.setOccupancy(originPacked, 2);

    const result = resolveDisplacement(
      { congestedCell: origin },
      capacityEvaluator,
      occupancyTracker
    );

    expect(result.ok).toBe(true);
    if (result.ok) {
      expect(result.targetPacked).toBe(originPacked);
      expect(result.traversalCostUnits).toBe(0);
      expect(result.traversalPath.length).toBe(1);
    }
  });

  it('routes displaced entity to adjacent cell with available capacity', () => {
    const origin = createAxialCoord(0, 0);
    const originPacked = packAxial(0, 0);

    // 3 occupants in cell with capacity 2 -> Eviction required
    occupancyTracker.setOccupancy(originPacked, 3);

    const result = resolveDisplacement(
      { congestedCell: origin },
      capacityEvaluator,
      occupancyTracker
    );

    expect(result.ok).toBe(true);
    if (result.ok) {
      expect(result.targetPacked).not.toBe(originPacked);
      expect(result.traversalCostUnits).toBe(100); // 1 step to neighbor
      expect(result.traversalPath.length).toBe(2);
    }
  });

  it('routes around saturated neighbors and walls without teleportation', () => {
    const origin = createAxialCoord(0, 0);
    const originPacked = packAxial(0, 0);
    occupancyTracker.setOccupancy(originPacked, 3);

    // Block 5 neighbors with walls or full capacity
    const blocked = new Set<PackedAxial>([
      packAxial(1, 0),
      packAxial(1, -1),
      packAxial(0, -1),
      packAxial(-1, 0),
      packAxial(-1, 1)
    ]);

    // Only neighbor (0, 1) is open, but saturate it to capacity 2
    const nOpen = packAxial(0, 1);
    occupancyTracker.setOccupancy(nOpen, 2);

    // Cell (0, 2) is open and empty
    const result = resolveDisplacement(
      {
        congestedCell: origin,
        blockedCells: blocked,
        budgetCostUnits: 500
      },
      capacityEvaluator,
      occupancyTracker
    );

    expect(result.ok).toBe(true);
    if (result.ok) {
      // Must step through (0, 1) to reach (0, 2)
      expect(result.targetPacked).toBe(packAxial(0, 2));
      expect(result.traversalCostUnits).toBe(200);
      expect(result.traversalPath.length).toBe(3);
    }
  });

  it('halts with BUDGET_EXCEEDED when search expands beyond 512 nodes', () => {
    const origin = createAxialCoord(0, 0);
    const originPacked = packAxial(0, 0);
    occupancyTracker.setOccupancy(originPacked, 3);

    // Satiate everything within radius so search expands indefinitely until envelope ceiling
    const capacityAlwaysFull = createCapacityEvaluator({ cellCapacityDefault: 0 });

    const result = resolveDisplacement(
      {
        congestedCell: origin,
        budgetCostUnits: 100000,
        workEnvelopeLimit: 512
      },
      capacityAlwaysFull,
      occupancyTracker
    );

    expect(result.ok).toBe(false);
    if (!result.ok) {
      expect(result.reason).toBe('BUDGET_EXCEEDED');
      expect(result.expandedSteps).toBe(513);
    }
  });

  it('dispatches DIS01 Blessed menu action within 2ms execution budget', async () => {
    const action = createDisplacementAction();
    const bus = createSpatialBus();
    const logs: string[] = [];

    bus.subscribe('TELEMETRY_LOG', (evt) => {
      logs.push(String(evt.payload.message || ''));
    });

    const start = performance.now();
    await action.execute(bus);
    const elapsed = performance.now() - start;

    expect(elapsed).toBeLessThanOrEqual(2.0);
    expect(logs.some((l) => l.includes('[DIS01] Overcrowding Alert'))).toBe(true);
    expect(logs.some((l) => l.includes('Eviction Resolved'))).toBe(true);
  });
});
```
