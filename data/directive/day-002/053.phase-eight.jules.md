export function deepFreeze<T>(obj: T): T {
  if (obj === null || typeof obj !== 'object') return obj;
  Object.freeze(obj);
  for (const key of Object.getOwnPropertyNames(obj)) {
    const val = (obj as any)[key];
    if (val !== null && typeof val === 'object' && !Object.isFrozen(val)) {
      deepFreeze(val);
    }
  }
  return obj;
}

export function canonicalizeJson(obj: unknown): string {
  if (obj === null) return 'null';
  if (typeof obj === 'number') {
    if (!Number.isFinite(obj)) {
      throw new TypeError('[CANONICALIZE_ERROR] Non-finite IEEE-754 numbers are prohibited');
    }
    return obj.toString();
  }
  if (typeof obj === 'boolean') return obj ? 'true' : 'false';
  if (typeof obj === 'string') return JSON.stringify(obj.normalize('NFC'));
  if (Array.isArray(obj)) return `[${obj.map((el) => canonicalizeJson(el)).join(',')}]`;
  if (typeof obj === 'object') {
    const sortedKeys = Object.keys(obj as Record<string, unknown>).sort();
    const entries = sortedKeys.map(
      (k) => `${JSON.stringify(k.normalize('NFC'))}:${canonicalizeJson((obj as any)[k])}`
    );
    return `{${entries.join(',')}}`;
  }
  throw new TypeError(`[CANONICALIZE_ERROR] Unsupported datatype: ${typeof obj}`);
}

export async function computeSha256Hex(content: string): Promise<string> {
  const enc = new TextEncoder();
  const data = enc.encode(content);
  const hashBuf = await crypto.subtle.digest('SHA-256', data);
  const byteArr = Array.from(new Uint8Array(hashBuf));
  return byteArr.map((b) => b.toString(16).padStart(2, '0')).join('');
}

export function sanitizeQueryPayload(raw: unknown): {
  valid: boolean;
  query?: any;
  strippedKeys: string[];
  breachReason?: string;
} {
  if (raw === null || typeof raw !== 'object') {
    return { valid: false, strippedKeys: [], breachReason: '[SCHEMA_BREACH] Payload must be an object' };
  }

  const strippedKeys: string[] = [];
  const cloned = JSON.parse(JSON.stringify(raw));

  function scrub(node: Record<string, any>) {
    for (const k of Object.keys(node)) {
      if (BANNED_NARRATIVE_TAGS.includes(k)) {
        delete node[k];
        strippedKeys.push(k);
      } else if (typeof node[k] === 'object' && node[k] !== null) {
        scrub(node[k]);
      }
    }
  }
  scrub(cloned);

  if (!cloned.queryType) {
    return { valid: false, strippedKeys, breachReason: '[SCHEMA_BREACH] Missing queryType' };
  }

  if (cloned.origin) {
    const checkOrigin = EdgeAxialCoordSchema.safeParse(cloned.origin);
    if (!checkOrigin.success) {
      return { valid: false, strippedKeys, breachReason: checkOrigin.error.errors[0].message };

  if (cloned.target) {
    const checkTarget = EdgeAxialCoordSchema.safeParse(cloned.target);
    if (!checkTarget.success) {
      return { valid: false, strippedKeys, breachReason: checkTarget.error.errors[0].message };

    const checkDest = EdgeAxialCoordSchema.safeParse(cloned.destination);
    if (!checkDest.success) {
      return { valid: false, strippedKeys, breachReason: checkDest.error.errors[0].message };
    }
  }

  return { valid: true, query: cloned, strippedKeys };
}

    const elapsed = Math.round((performance.now() - start) * 1000);
    const avg = queryKeys.length > 0 ? Math.round(elapsed / queryKeys.length) : 0;
    const batchAuditHash = await computeSha256Hex(canonicalizeJson(auditMap));

    return deepFreeze({
      ok: failedCount === 0,
      batchId: envelope.batchId,
      summary: {
        totalQueries: queryKeys.length,
        succeededCount,
        failedCount,
        totalDurationMicros: elapsed,
        averageDurationMicros: avg,
        batchAuditHash,
      },
      receipts,
    });
  }
}
