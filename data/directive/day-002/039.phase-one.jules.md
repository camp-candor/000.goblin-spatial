export const AXIAL_COORD_MAX = 32767;

/**
 * Strict 2D Axial Hex Coordinate Schema.
 * Enforces exact integer values within [-32768, 32767] and planar cubic invariant q + r + s === 0.
 */
export const AxialCoordSchema = z
  .object({
    q: z
      .number()
      .int({ message: '[FLOAT_PROHIBITED] Coordinate q must be an integer' })
      .min(AXIAL_COORD_MIN, { message: '[OUT_OF_BOUNDS] Coordinate q underflows -32768' })
      .max(AXIAL_COORD_MAX, { message: '[OUT_OF_BOUNDS] Coordinate q overflows 32767' }),
    r: z
      .number()
      .int({ message: '[FLOAT_PROHIBITED] Coordinate r must be an integer' })
      .min(AXIAL_COORD_MIN, { message: '[OUT_OF_BOUNDS] Coordinate r underflows -32768' })
      .max(AXIAL_COORD_MAX, { message: '[OUT_OF_BOUNDS] Coordinate r overflows 32767' }),
    s: z
      .number()
      .int({ message: '[FLOAT_PROHIBITED] Coordinate s must be an integer' })
      .min(AXIAL_COORD_MIN * 2, { message: '[OUT_OF_BOUNDS] Coordinate s underflows' })
      .max(AXIAL_COORD_MAX * 2, { message: '[OUT_OF_BOUNDS] Coordinate s overflows' }),
  })
  .strict()
  .refine((data) => data.q + data.r + data.s === 0, {
    message: '[CUBIC_INVARIANT_BREACH] Planar coordinates must satisfy q + r + s === 0',
  });

export type AxialCoord = z.infer<typeof AxialCoordSchema>;

/**
 * Branded 32-bit unsigned integer packed axial cell identifier.
 */
export type PackedAxial = number & { readonly __brand: unique symbol };

export function packAxialCell(q: number, r: number): PackedAxial {
  if (!Number.isInteger(q) || !Number.isInteger(r)) {
    throw new TypeError('[FLOAT_PROHIBITED] Coordinates must be exact integers');
  }
  if (q < AXIAL_COORD_MIN || q > AXIAL_COORD_MAX || r < AXIAL_COORD_MIN || r > AXIAL_COORD_MAX) {
    throw new RangeError('[OUT_OF_BOUNDS] Coordinate outside [-32768, 32767]');
  }
  const qOffset = (q + 32768) & 0xffff;
  const rOffset = (r + 32768) & 0xffff;
  return (((qOffset << 16) | rOffset) >>> 0) as PackedAxial;
}

export function unpackAxialCell(packed: number): AxialCoord {
  if (!Number.isInteger(packed) || packed < 0 || packed > 0xffffffff) {
    throw new TypeError('[INVALID_PACKED_CELL] PackedAxial must be an unsigned 32-bit integer');
  }
  const qOffset = (packed >>> 16) & 0xffff;
  const rOffset = packed & 0xffff;
  const q = qOffset - 32768;
  const r = rOffset - 32768;
  const s = -q - r;
  return Object.freeze({ q, r, s });
}

import type { TopologyManifest } from '../fce/manifest.schema.js';
import type { DynamicOverlay, MergedTopologySnapshot } from '../fce/overlay.schema.js';

/**
 * Pure in-memory overlay merger.
 * Superimposes dynamic deltas onto the static topology manifest without mutating source records.
 */
export function mergeOverlaySnapshot(
  manifest: TopologyManifest,
  overlay: DynamicOverlay
): MergedTopologySnapshot {
  const mergedPortals = manifest.portals.map((portal) => {
    const isOverridden = Object.prototype.hasOwnProperty.call(overlay.portalOverrides, portal.portalId);
    const traversable = isOverridden ? overlay.portalOverrides[portal.portalId] : portal.traversableDefault;
    return Object.freeze({
      portalId: portal.portalId,
      fromRoomId: portal.fromRoomId,
      toRoomId: portal.toRoomId,
      entryCoord: Object.freeze({ ...portal.entryCoord }),
      exitCoord: Object.freeze({ ...portal.exitCoord }),
      acousticDampingCdb: portal.acousticDampingCdb,
      traversable,
    });
  });

  const mergedOccluders = manifest.occluders.map((occ) => {
    const hasLoSOverride = Object.prototype.hasOwnProperty.call(overlay.occluderOverrides, occ.occluderId);
    const blocksLoS = hasLoSOverride ? overlay.occluderOverrides[occ.occluderId] : occ.blocksLoSDefault;

    const deltaCdb = overlay.attenuationDeltasCdb[occ.occluderId] ?? 0;
    const finalAttenuation = Math.max(0, occ.attenuationCdb + deltaCdb);

    return Object.freeze({
      occluderId: occ.occluderId,
      coord: Object.freeze({ ...occ.coord }),
      blocksLoS,
      attenuationCdb: finalAttenuation,
    });
  });

  const mergedCells = manifest.cells.map((cell) =>
    Object.freeze({
      q: cell.coord.q,
      r: cell.coord.r,
      s: cell.coord.s,
      costUnits: cell.costUnits,
      passable: cell.passable,
    })
  );

  return Object.freeze({
    topologyId: manifest.topologyId,
    overlayId: overlay.overlayId,
    cells: Object.freeze(mergedCells),
    portals: Object.freeze(mergedPortals),
    occluders: Object.freeze(mergedOccluders),
  });
}
