import { SpatialBus, SpatialBusEvent, SpatialBusListener } from '../../98.menu.unit/fce/menu.interface.js';

export class ReactiveSpatialBus implements SpatialBus {
  private readonly listeners: Map<string, Set<SpatialBusListener>> = new Map();
  private history: SpatialBusEvent[] = [];

  public emit(type: string, payload: Record<string, unknown> = {}): void {
    const event: SpatialBusEvent = {
      type,
      payload,
      timestamp: Date.now()
    };
    this.history.push(event);

    const typeListeners = this.listeners.get(type);
    if (typeListeners) {
      for (const listener of typeListeners) {
        try {
          listener(event);
        } catch (err) {
          // Isolate subscriber exceptions to prevent event loop collapse
          const errorMsg = err instanceof Error ? err.message : String(err);
          process.stderr.write(`:: [BUS_ERROR] [${type}] ${errorMsg}\n`);
        }
      }
    }
  }

  public subscribe(type: string, listener: SpatialBusListener): () => void {
    if (!this.listeners.has(type)) {
      this.listeners.set(type, new Set());
    }
    const set = this.listeners.get(type)!;
    set.add(listener);

    return () => {
      set.delete(listener);
      if (set.size === 0) {
        this.listeners.delete(type);
      }
    };
  }

  public getHistory(): readonly SpatialBusEvent[] {
    return this.history;
  }

  public clear(): void {
    this.listeners.clear();
    this.history = [];
  }
}

export const createSpatialBus = (): SpatialBus => new ReactiveSpatialBus();

import { MenuActionPayload, MenuCatalogState, SpatialBus } from '../fce/menu.interface.js';

export class MenuDispatcher {
  private actions: MenuActionPayload[] = [];
  private activeIndex: number = 0;

  constructor(private readonly bus: SpatialBus) {
    this.registerDefaultActions();
  }

  public registerAction(action: MenuActionPayload): void {
    const exists = this.actions.some((a) => a.id === action.id);
    if (exists) {
      throw new Error(`DUPLICATE_ACTION_ID: ${action.id}`);
    }
    this.actions.push(action);
  }

  public getCatalog(): MenuCatalogState {
    const currentAction = this.actions[this.activeIndex];
    return {
      activeIndex: this.activeIndex,
      activeCategory: currentAction ? currentAction.category : 'NONE',
      actions: [...this.actions]
    };
  }

  public selectNext(): void {
    if (this.actions.length === 0) return;
    this.activeIndex = (this.activeIndex + 1) % this.actions.length;
    this.bus.emit('MENU_INDEX_CHANGED', { index: this.activeIndex });
  }

  public selectPrevious(): void {
    if (this.actions.length === 0) return;
    this.activeIndex = (this.activeIndex - 1 + this.actions.length) % this.actions.length;
    this.bus.emit('MENU_INDEX_CHANGED', { index: this.activeIndex });
  }

  public async executeSelected(): Promise<void> {
    const action = this.actions[this.activeIndex];
    if (!action) {
      throw new Error('NO_ACTION_SELECTED');
    }
    this.bus.emit('TELEMETRY_LOG', { message: `>> EXECUTING: [${action.shortcut}] ${action.label}` });
    await action.execute(this.bus);
    this.bus.emit('TELEMETRY_LOG', { message: `>> COMPLETED: [${action.shortcut}] ${action.label}` });
  }

  private registerDefaultActions(): void {
    this.registerAction({
      id: 'D01_RESET_KERNEL',
      label: 'Reset Spatial Kernel State',
      shortcut: 'D01',
      category: 'DIAGNOSTIC',
      execute: async (bus) => {
        bus.emit('KERNEL_STATE_RESET', { timestamp: Date.now() });
      }
    });

    this.registerAction({
      id: 'D02_INSPECT_TOPOLOGY',
      label: 'Inspect Active Topology Manifest',
      shortcut: 'D02',
      category: 'TOPOLOGY',
      execute: async (bus) => {
        bus.emit('TELEMETRY_LOG', { message: ':: Manifest: [000.goblin-spatial] Cached Nodes: 0' });
      }
    });

    this.registerAction({
      id: 'D03_RUN_BENCHMARK',
      label: 'Execute Sub-Millisecond Profiler',
      shortcut: 'D03',
      category: 'SIMULATION',
      execute: async (bus) => {
        bus.emit('TELEMETRY_LOG', { message: ':: Benchmark: Budget Target <= 2ms. Status: NOMINAL' });
      }
    });
  }
}

import { createSpatialBus } from '../99.bus.unit/buz/bus.buzz.js';
import { MenuDispatcher } from '../98.menu.unit/buz/00.menu.buzz.js';

export interface CockpitInstance {
  readonly bus: ReturnType<typeof createSpatialBus>;
  readonly menu: MenuDispatcher;
  readonly renderTelemetryFrame: () => string;
  destroy(): void;
}

export function initializeCockpitHarness(): CockpitInstance {
  const bus = createSpatialBus();
  const menu = new MenuDispatcher(bus);
  const logs: string[] = [];

  const unsubscribe = bus.subscribe('TELEMETRY_LOG', (evt) => {
    const msg = String(evt.payload.message || '');
    logs.push(msg);
    if (logs.length > 100) {
      logs.shift();
    }
  });

  const renderTelemetryFrame = (): string => {
    const catalog = menu.getCatalog();
    const actionLines = catalog.actions
      .map((act, i) => {
        const marker = i === catalog.activeIndex ? '>' : ' ';
        return `  ${marker} [${act.shortcut}] ${act.label} (${act.category})`;
      })
      .join('\n');

    const recentLogs = logs.slice(-5).join('\n') || '  >> [SYSTEM READY: Awaiting Operator Input]';

    return [
      '+--------------------------------------------------------------------------+',
      '| 000.goblin-spatial :: BLESSED TERMINAL COCKPIT (Phase 1)                |',
      '+--------------------------------------------------------------------------+',
      '| MENU CONTROLS (Columns 0-4, Rows 0-12)                                   |',
      actionLines,
      '+--------------------------------------------------------------------------+',
      '| TELEMETRY LOG [cns00] (Columns 4-12, Rows 3-12)                          |',
      recentLogs,
      '+--------------------------------------------------------------------------+'
    ].join('\n');
  };

  return {
    bus,
    menu,
    renderTelemetryFrame,
    destroy: () => {
      unsubscribe();
      bus.clear();
    }
  };
}

if (process.env.NODE_ENV !== 'test' && typeof process.stdout.write === 'function') {
  const cockpit = initializeCockpitHarness();
  process.stdout.write(cockpit.renderTelemetryFrame() + '\n');
}
