import { AxialCoord, createAxialCoord, PackedAxial, packAxial } from '../../28.axial.unit/val/axial.model.js';
import { axialToDoubled } from '../../28.axial.unit/buz/axial.buzz.js';
import { VertexKey } from '../fce/observer.interface.js';

/**
 * 3-Hex Vertex Majority Truth Table:
 * Evaluates whether a ray grazing a shared vertex between Hexes (U, V, W) is severed.
 * Index = (U_opaque << 2) | (V_opaque << 1) | W_opaque
 * Blocked iff sum(U, V, W) >= 2 (Majority Rule).
 */
export const VERTEX_MAJORITY_TABLE: readonly boolean[] = [
  false, // 000: 0 opaque -> Clear
  false, // 001: 1 opaque -> Clear (grazes corner of 1 wall)
  false, // 010: 1 opaque -> Clear
  true,  // 011: 2 opaque -> Blocked (pinched between 2 walls)
  false, // 100: 1 opaque -> Clear
  true,  // 101: 2 opaque -> Blocked
  true,  // 110: 2 opaque -> Blocked
  true   // 111: 3 opaque -> Blocked (completely embedded)
] as const;

export function evaluateVertexMajorityMask(mask: number): boolean {
  if (mask < 0 || mask > 7) {
    throw new RangeError(`VERTEX_MASK_OUT_OF_BOUNDS: ${mask}`);
  }
  return VERTEX_MAJORITY_TABLE[mask];
}

/**
 * In doubled coordinate space (cx = 4q + 2r, cy = 6r):
 * Vertices are located at (cx, cy +/- 4) and (cx +/- 2, cy +/- 2).
 * Every shared vertex meets exactly three adjacent hexagons.
 */
export function getThreeHexesAtVertex(
  vx: bigint,
  vy: bigint
): readonly [AxialCoord, AxialCoord, AxialCoord] {
  // Case A: Vertex with cy = 6r + 4 or 6r - 2
  // We identify the three surrounding axial coordinate cells deterministically
  const xNum = Number(vx);
  const yNum = Number(vy);

  // A vertex (vx, vy) in doubled pointy-top hex tiling belongs to:
  // Hex 1: closest cell with y mod 6 === 0
  const candidateOffsets: readonly { readonly dq: number; readonly dr: number }[] = [
    { dq: 0, dr: 0 },
    { dq: 0, dr: 1 },
    { dq: -1, dr: 1 },
    { dq: 1, dr: 0 },
    { dq: 0, dr: -1 },
    { dq: 1, dr: -1 },
    { dq: -1, dr: 0 }
  ];

  const matchedCells: AxialCoord[] = [];
  const baseR = Math.floor(yNum / 6);
  const baseQ = Math.floor((xNum - 2 * baseR) / 4);

  for (let dq = -2; dq <= 2; dq++) {
    for (let dr = -2; dr <= 2; dr++) {
      const q = baseQ + dq;
      const r = baseR + dr;
      const d = axialToDoubled(createAxialCoord(q, r));
      const cx = BigInt(d.x);
      const cy = BigInt(d.y);

      // Check if (vx, vy) is one of the 6 vertices of this cell
      const isVertex =
        (vx === cx && (vy === cy + 4n || vy === cy - 4n)) ||
        ((vx === cx + 2n || vx === cx - 2n) && (vy === cy + 2n || vy === cy - 2n));

      if (isVertex) {
        matchedCells.push(createAxialCoord(q, r));
      }
    }
  }

  // Deterministically sort by packed cell ID ascending
  matchedCells.sort((a, b) => packAxial(a.q, a.r) - packAxial(b.q, b.r));

  if (matchedCells.length !== 3) {
    throw new Error(
      `VERTEX_TOPOLOGY_DEFECT: Expected exactly 3 hexes at vertex (${vx}, ${vy}), found ${matchedCells.length}`
    );
  }

  return [matchedCells[0], matchedCells[1], matchedCells[2]];
}

import { AxialCoord, PackedAxial, packAxial } from '../../28.axial.unit/val/axial.model.js';
import { axialToDoubled } from '../../28.axial.unit/buz/axial.buzz.js';
import {
  VertexKey,
  SharedVertexEvent,
  VertexPolicyEvaluation
} from '../fce/observer.interface.js';
import {
  evaluateVertexMajorityMask,
  getThreeHexesAtVertex
} from '../val/observer.model.js';

export function crossProduct2D(
  p1x: bigint,
  p1y: bigint,
  p2x: bigint,
  p2y: bigint,
  px: bigint,
  py: bigint
): bigint {
  return (p2x - p1x) * (py - p1y) - (p2y - p1y) * (px - p1x);
}

export function isPointOnFiniteSegment(
  p1x: bigint,
  p1y: bigint,
  p2x: bigint,
  p2y: bigint,
  px: bigint,
  py: bigint
): boolean {
  if (crossProduct2D(p1x, p1y, p2x, p2y, px, py) !== 0n) {
    return false;
  }
  const minX = p1x < p2x ? p1x : p2x;
  const maxX = p1x > p2x ? p1x : p2x;
  const minY = p1y < p2y ? p1y : p2y;
  const maxY = p1y > p2y ? p1y : p2y;

  return px >= minX && px <= maxX && py >= minY && py <= maxY;
}

/**
 * Evaluates the 3-Hex Majority Rule for all vertices intersected by segment AB.
 * Origin and target endpoints are strictly exempt (treated as transparent).
 */
export function evaluateRayVertexCrossings(
  origin: AxialCoord,
  target: AxialCoord,
  isCellOpaque: (packed: PackedAxial) => boolean
): VertexPolicyEvaluation {
  const originPacked = packAxial(origin.q, origin.r);
  const targetPacked = packAxial(target.q, target.r);

  if (originPacked === targetPacked) {
    return { hasClearLineOfSight: true, vertexEvents: [] };
  }

  const dA = axialToDoubled(origin);
  const dB = axialToDoubled(target);
  const p1x = BigInt(dA.x);
  const p1y = BigInt(dA.y);
  const p2x = BigInt(dB.x);
  const p2y = BigInt(dB.y);

  const minQ = Math.min(origin.q, target.q) - 1;
  const maxQ = Math.max(origin.q, target.q) + 1;
  const minR = Math.min(origin.r, target.r) - 1;
  const maxR = Math.max(origin.r, target.r) + 1;

  const testedVertices = new Map<string, VertexKey>();

  for (let q = minQ; q <= maxQ; q++) {
    for (let r = minR; r <= maxR; r++) {
      const d = axialToDoubled({ q, r, s: -q - r });
      const cx = BigInt(d.x);
      const cy = BigInt(d.y);

      const vertices: readonly VertexKey[] = [
        { vx: cx, vy: cy + 4n },
        { vx: cx + 2n, vy: cy + 2n },
        { vx: cx + 2n, vy: cy - 2n },
        { vx: cx, vy: cy - 4n },
        { vx: cx - 2n, vy: cy - 2n },
        { vx: cx - 2n, vy: cy + 2n }
      ];

      for (const v of vertices) {
        const key = `${v.vx},${v.vy}`;
        if (!testedVertices.has(key)) {
          if (isPointOnFiniteSegment(p1x, p1y, p2x, p2y, v.vx, v.vy)) {
            testedVertices.set(key, v);
          }
        }
      }
    }
  }

  const vertexEvents: SharedVertexEvent[] = [];

  for (const v of testedVertices.values()) {
    const hexes = getThreeHexesAtVertex(v.vx, v.vy);
    const packedHexes: [PackedAxial, PackedAxial, PackedAxial] = [
      packAxial(hexes[0].q, hexes[0].r),
      packAxial(hexes[1].q, hexes[1].r),
      packAxial(hexes[2].q, hexes[2].r)
    ];

    // Endpoint Exemption: Origin and target count strictly as transparent (0)
    const uOpaque = packedHexes[0] !== originPacked && packedHexes[0] !== targetPacked && isCellOpaque(packedHexes[0]) ? 1 : 0;
    const vOpaque = packedHexes[1] !== originPacked && packedHexes[1] !== targetPacked && isCellOpaque(packedHexes[1]) ? 1 : 0;
    const wOpaque = packedHexes[2] !== originPacked && packedHexes[2] !== targetPacked && isCellOpaque(packedHexes[2]) ? 1 : 0;

    const mask = (uOpaque << 2) | (vOpaque << 1) | wOpaque;
    const isBlocked = evaluateVertexMajorityMask(mask);

    const event: SharedVertexEvent = {
      vertex: v,
      cells: hexes,
      packedCells: packedHexes,
      opaqueMask: mask,
      isBlocked
    };
    vertexEvents.push(event);

    if (isBlocked) {
      return {
        hasClearLineOfSight: false,
        vertexEvents,
        severedAtVertex: v
      };
    }
  }

  return {
    hasClearLineOfSight: true,
    vertexEvents
  };
}
