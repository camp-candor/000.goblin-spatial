```typescript
export const COORD_OFFSET = 32768; // Lattice domain clamped to [-32768, 32767]

export type PackedAxial = number & { readonly __brand: unique symbol };

export interface AxialCoord {
  readonly q: number;
  readonly r: number;
  readonly s: number; // Planar invariant: q + r + s === 0
}

export interface DoubledCoord {
  readonly x: number; // cx = 4q + 2r
  readonly y: number; // cy = 6r
}

export interface IntVector2D {
  readonly x: bigint;
  readonly y: bigint;
}

export function createAxialCoord(q: number, r: number): AxialCoord {
  if (!Number.isInteger(q) || !Number.isInteger(r)) {
    throw new TypeError(`AXIAL_NON_INTEGER_COORDINATE: (${q}, ${r})`);
  }
  if (q < -COORD_OFFSET || q >= COORD_OFFSET || r < -COORD_OFFSET || r >= COORD_OFFSET) {
    throw new RangeError(`AXIAL_COORDINATE_OUT_OF_BOUNDS: (${q}, ${r})`);
  }
  const s = -q - r;
  return { q, r, s };
}

export function packAxial(q: number, r: number): PackedAxial {
  if (!Number.isInteger(q) || !Number.isInteger(r)) {
    throw new TypeError(`PACK_NON_INTEGER: (${q}, ${r})`);
  }
  if (q < -COORD_OFFSET || q >= COORD_OFFSET || r < -COORD_OFFSET || r >= COORD_OFFSET) {
    throw new RangeError(`PACK_OUT_OF_BOUNDS: (${q}, ${r})`);
  }
  const qOffset = (q + COORD_OFFSET) & 0xFFFF;
  const rOffset = (r + COORD_OFFSET) & 0xFFFF;
  return (((qOffset << 16) | rOffset) >>> 0) as PackedAxial;
}

export function unpackAxial(packed: PackedAxial): AxialCoord {
  const u32 = packed >>> 0;
  const q = ((u32 >>> 16) & 0xFFFF) - COORD_OFFSET;
  const r = (u32 & 0xFFFF) - COORD_OFFSET;
  const s = -q - r;
  return { q, r, s };
}

export function unpackQ(packed: PackedAxial): number {
  return ((packed >>> 16) & 0xFFFF) - COORD_OFFSET;
}

export function unpackR(packed: PackedAxial): number {
  return (packed & 0xFFFF) - COORD_OFFSET;
}
```

```typescript
import {

  DoubledCoord,
  IntVector2D,
  createAxialCoord

export function assertAxialInvariant(coord: AxialCoord): void {
  if (coord.q + coord.r + coord.s !== 0) {
    throw new Error(`AXIAL_INVARIANT_VIOLATION: q(${coord.q}) + r(${coord.r}) + s(${coord.s}) !== 0`);
  }
}

/**
 * Doubled Cartesian Coordinate Mapping:
 * Circumradius s = 2 pointy-top hex projected via affine stretch:
 * Centroid(q, r) = (4q + 2r, 6r)
 */
export function axialToDoubled(coord: AxialCoord): DoubledCoord {
  assertAxialInvariant(coord);
  return {
    x: 4 * coord.q + 2 * coord.r,
    y: 6 * coord.r
  };
}

export function doubledToAxial(doubled: DoubledCoord): AxialCoord {
  if (!Number.isInteger(doubled.x) || !Number.isInteger(doubled.y)) {
    throw new TypeError(`DOUBLED_NON_INTEGER_COORDINATE: (${doubled.x}, ${doubled.y})`);
  }
  if (doubled.y % 6 !== 0) {
    throw new Error(`DOUBLED_INVALID_Y_ALIGNMENT: y(${doubled.y}) must be divisible by 6`);
  }
  const r = Math.trunc(doubled.y / 6);
  const remX = doubled.x - 2 * r;
  if (remX % 4 !== 0) {
    throw new Error(`DOUBLED_INVALID_X_ALIGNMENT: x(${doubled.x}) - 2r(${2 * r}) must be divisible by 4`);
  }
  const q = Math.trunc(remX / 4);
  return createAxialCoord(q, r);
}

/**
 * Metric Hexagonal Axial Distance:
 * d_axial(a, b) = max(|qa - qb|, |ra - rb|, |sa - sb|)
 */
export function axialDistance(a: AxialCoord, b: AxialCoord): number {
  assertAxialInvariant(a);
  assertAxialInvariant(b);
  const dq = Math.abs(a.q - b.q);
  const dr = Math.abs(a.r - b.r);
  const ds = Math.abs(a.s - b.s);
  return Math.max(dq, dr, ds);
}

/**
 * Exact Integer Chord Distance Squared:
 * Used for deterministic tie-breaking without square roots.
 */
export function chordDistanceSquared(a: AxialCoord, b: AxialCoord): bigint {
  assertAxialInvariant(a);
  assertAxialInvariant(b);
  const da = axialToDoubled(a);
  const db = axialToDoubled(b);
  const dx = BigInt(da.x - db.x);
  const dy = BigInt(da.y - db.y);
  return dx * dx + dy * dy;
}

export function hexVertices(coord: AxialCoord): readonly IntVector2D[] {
  const d = axialToDoubled(coord);
  const cx = BigInt(d.x);
  const cy = BigInt(d.y);
  return [
    { x: cx, y: cy + 4n },
    { x: cx + 2n, y: cy + 2n },
    { x: cx + 2n, y: cy - 2n },
    { x: cx, y: cy - 4n },
    { x: cx - 2n, y: cy - 2n },
    { x: cx - 2n, y: cy + 2n }
  ];
}
```

```typescript
import { AxialCoord, createAxialCoord } from '../../28.axial.unit/val/axial.model.js';

/**
 * Six Axial Directions ordered clockwise starting East (0 to 5):
 * 0: North-East (q: 0, r: -1)
 * 1: East       (q: 1, r: -1)
 * 2: South-East (q: 1, r: 0)
 * 3: South-West (q: 0, r: 1)
 * 4: West       (q: -1, r: 1)
 * 5: North-West (q: -1, r: 0)
 */
export const AXIAL_DIRECTIONS: readonly { readonly q: number; readonly r: number }[] = [
  { q: 0, r: -1 },
  { q: 1, r: -1 },
  { q: 1, r: 0 },
  { q: 0, r: 1 },
  { q: -1, r: 1 },
  { q: -1, r: 0 }
] as const;

export function getNeighbor(coord: AxialCoord, directionIndex: number): AxialCoord {
  assertAxialInvariant(coord);
  const idx = ((directionIndex % 6) + 6) % 6;
  const dir = AXIAL_DIRECTIONS[idx];
  return createAxialCoord(coord.q + dir.q, coord.r + dir.r);
}

export function getAllNeighbors(coord: AxialCoord): readonly AxialCoord[] {
  assertAxialInvariant(coord);
  const out: AxialCoord[] = [];
  for (let i = 0; i < 6; i++) {
    const dir = AXIAL_DIRECTIONS[i];
    out.push(createAxialCoord(coord.q + dir.q, coord.r + dir.r));
  }
  return out;
}

  assertAxialInvariant(coord);

    const dir = AXIAL_DIRECTIONS[i];
```

```typescript
import { AxialCoord, createAxialCoord } from '../../28.axial.unit/val/axial.model.js';
import { assertAxialInvariant, axialDistance } from '../../28.axial.unit/buz/axial.buzz.js';
import { AXIAL_DIRECTIONS, getNeighbor } from '../../46.neighbor.unit/buz/neighbor.buzz.js';

export function getRing(center: AxialCoord, radius: number): readonly AxialCoord[] {
  assertAxialInvariant(center);
  if (!Number.isInteger(radius) || radius < 0) {
    throw new RangeError(`RING_INVALID_RADIUS: radius must be non-negative integer (${radius})`);
  }
  if (radius === 0) {
    return [center];
  }

  const results: AxialCoord[] = [];
  // Start radius steps in direction 4 (West)
  let current = createAxialCoord(
    center.q + AXIAL_DIRECTIONS[4].q * radius,
    center.r + AXIAL_DIRECTIONS[4].r * radius
  );

  for (let dirIndex = 0; dirIndex < 6; dirIndex++) {
    for (let step = 0; step < radius; step++) {
      results.push(current);
      current = getNeighbor(current, dirIndex);
    }
  }

  return results;
}

export function getSpiralDisk(center: AxialCoord, radius: number): readonly AxialCoord[] {
  assertAxialInvariant(center);
  if (!Number.isInteger(radius) || radius < 0) {
    throw new RangeError(`DISK_INVALID_RADIUS: radius must be non-negative integer (${radius})`);
  }

  const disk: AxialCoord[] = [center];
  for (let r = 1; r <= radius; r++) {
    const ring = getRing(center, r);
    for (let i = 0; i < ring.length; i++) {
      disk.push(ring[i]);
    }
  }

  // Deterministic sort: distance ascending, then q ascending, then r ascending
  return disk.sort((a, b) => {
    const distA = axialDistance(center, a);
    const distB = axialDistance(center, b);
    if (distA !== distB) return distA - distB;
    if (a.q !== b.q) return a.q - b.q;
    return a.r - b.r;
  });
}

export function expectedHexCountForRadius(radius: number): number {
  if (radius < 0) return 0;
  return 3 * radius * (radius + 1) + 1;
}
```
