function sanitizeAscii(raw: string): string {
  return String(raw || '')
    .replace(/[^\x20-\x7E]/g, '')
    .trim()
}

async function emitSentinelTelemetry(
  ste: State | undefined,
  lines: readonly string[],
): Promise<void> {
  const lib = (globalThis as any).LIBRARY || (global as any).LIBRARY
  for (const raw of lines) {
    const clean = sanitizeAscii(raw)
    if (lib && typeof lib.hunt === 'function') {
      await lib
        .hunt(UPDATE_CONSOLE, { idx: 'cns00', src: clean })
        .catch(() => {})
    } else if (ste && typeof ste.hunt === 'function') {
      await ste
        .hunt(UPDATE_CONSOLE, { idx: 'cns00', src: clean })
        .catch(() => {})
    }
  }
}

export function resolveSpatialPackageDirectory(
  startDir: string = process.cwd(),
): string {
  const resolvedStart = path.resolve(startDir)
  if (
    fs.existsSync(path.join(resolvedStart, 'BEE.ts')) &&
    fs.existsSync(path.join(resolvedStart, '00.spatial.unit'))
  ) {
    return resolvedStart
  }
  const candidate = path.join(resolvedStart, 'packages', '301.spatial')
  if (fs.existsSync(path.join(candidate, 'BEE.ts'))) {
    return candidate
  }
  let curr = resolvedStart
  while (curr && curr !== path.dirname(curr)) {
    const nested = path.join(curr, 'packages', '301.spatial')
    if (fs.existsSync(path.join(nested, 'BEE.ts'))) {
      return nested
    }
    curr = path.dirname(curr)
  }
  return candidate
}

/**
 * Task 6.1: Pure BEE unit static self-verification sentinel auditing BEE.ts
 * unit registrations and stateful RESET_<UNIT> action/reducer wiring against
 * SPATIAL_RESET_MANIFEST.
 */
export async function auditSpatialResetCoverage(
  packageRootInput?: string,
  ste?: State,
): Promise<SpatialResetCoverageReport> {
  const packageRoot = resolveSpatialPackageDirectory(
    packageRootInput || process.cwd(),
  )
  const beePath = path.join(packageRoot, 'BEE.ts')
  const unhookedUnits: string[] = []
  const missingWiringDiagnostics: string[] = []

  const manifest = validateSpatialResetManifest(SPATIAL_RESET_MANIFEST)

  let mountedUnits: string[] = []
  if (fs.existsSync(beePath)) {
    const beeSrc = fs.readFileSync(beePath, 'utf8')
    const matches = [
      ...beeSrc.matchAll(/\.\/(\d{2}\.[a-z0-9-]+\.unit)\//g),
    ].map((m) => m[1])
    mountedUnits = [...new Set(matches)].sort((a, b) => a.localeCompare(b))
  } else {
    missingWiringDiagnostics.push(`Missing BEE.ts manifest at ${beePath}`)
  }

  const registryMap = new Map(
    manifest.unitRegistry.map((entry) => [entry.unitFolder, entry]),
  )

  for (const unitFolder of mountedUnits) {
    const descriptor = registryMap.get(unitFolder)
    if (!descriptor) {
      unhookedUnits.push(unitFolder)
      missingWiringDiagnostics.push(
        `Unregistered unit in BEE.ts: ${unitFolder} missing from SPATIAL_RESET_MANIFEST.unitRegistry`,
      )
      continue
    }

    if (descriptor.classification === 'STATEFUL_RESETTABLE') {
      const unitSlug = unitFolder.replace(/^\d{2}\./, '').replace(/\.unit$/, '')
      const actionPath = path.join(
        packageRoot,
        unitFolder,
        `${unitSlug}.action.ts`,
      )
      const reducePath = path.join(
        packageRoot,
        unitFolder,
        `${unitSlug}.reduce.ts`,
      )
      const expectedConstant = `RESET_${unitSlug.toUpperCase().replace(/-/g, '_')}`

      if (fs.existsSync(actionPath)) {
        const actionSrc = fs.readFileSync(actionPath, 'utf8')
        if (
          !actionSrc.includes(expectedConstant) &&
          descriptor.resetActionConstant &&
          !actionSrc.includes(descriptor.resetActionConstant)
        ) {
          unhookedUnits.push(unitFolder)
          missingWiringDiagnostics.push(
            `Stateful unit ${unitFolder} missing ${expectedConstant} in ${unitSlug}.action.ts`,
          )
        }
      }

      if (fs.existsSync(reducePath)) {
        const reduceSrc = fs.readFileSync(reducePath, 'utf8')
        if (!reduceSrc.includes(`Act.${expectedConstant}`)) {
          unhookedUnits.push(unitFolder)
          missingWiringDiagnostics.push(
            `Stateful unit ${unitFolder} missing Act.${expectedConstant} case in ${unitSlug}.reduce.ts`,
          )
        }
      }
    }
  }

  const statefulUnitCount = manifest.unitRegistry.filter(
    (u) => u.classification === 'STATEFUL_RESETTABLE',
  ).length
  const statelessUnitCount = manifest.unitRegistry.filter(
    (u) => u.classification === 'STATELESS_PURE',
  ).length

  const uniqueUnhooked = [...new Set(unhookedUnits)]
  const valid =
    mountedUnits.length > 0 &&
    uniqueUnhooked.length === 0 &&
    missingWiringDiagnostics.length === 0

  const receipts: string[] = [
    '>> ============================================================',
    '>> [SENTINEL:COVERAGE] SPATIAL TABULA RASA RESET COVERAGE AUDIT',
    `>> Package Root     : ${packageRoot}`,
    `>> Mounted Units    : ${mountedUnits.length} BEE.ts units scanned`,
    `>> Registry Split   : ${statefulUnitCount} stateful | ${statelessUnitCount} stateless`,
    `>> Unhooked Units   : ${uniqueUnhooked.length === 0 ? 'NONE [OK]' : uniqueUnhooked.join(', ')}`,
    `>> Sentinel Verdict : ${valid ? 'PASS [OK]' : 'FAIL [BREACH]'}`,
    '>> ============================================================',
  ]

  await emitSentinelTelemetry(ste, receipts)

  return {
    valid,
    packageRoot,
    mountedUnitCount: mountedUnits.length,
    statefulUnitCount,
    statelessUnitCount,
    unhookedUnits: uniqueUnhooked,
    missingWiringDiagnostics,
    receipts,
  }
}

      occupancy: occupancyModel,

          await resetTopology(topologyModel, bale, localBus)
          break

          await resetLedger(ledgerModel, bale, localBus)
          break

          await resetOccupancy(occupancyModel, bale, localBus)

          await resetEdge(edgeModel, bale, localBus)

          await resetCapacity(capacityModel, bale, localBus)
          break

          await resetOverlay(overlayModel, bale, localBus)

          await resetCollect(collectModel, bale, localBus)
          break
        case '[Menu action] Reset Menu':
          await resetMenu(menuModel, bale, localBus)
          break

      return {}
    },

  // Step 1: Verify initial Void_0 baseline

    spatialRoot,
    localBus,
    engineRoot,
  )
  const initialVoidVerified =
    initialVoid.isVoidState && initialVoid.residualDiskArtifacts === 0

      'spatial_proof_001.json',

  spatialModel.totalSynchronousInvocations = 42

  topologyModel.totalLoadedCells = 91

    spatialRoot,

  if (!zeroOrphanFilesVerified) {

      `Expected 0 residual disk artifacts, found ${postVoid.residualDiskArtifacts}`,

  const zeroModelContaminationVerified =
    spatialModel.totalSynchronousInvocations === 0 &&

    incidentModel.incidentReceiptCount === 0 &&
    witnessModel.stagedAppointmentProofs.length === 0 &&

    portModel.totalQueriesProcessed === 0 &&
    collectModel.totalBatchesProcessed === 0 &&
    menuModel.statusMessage === 'SPATIAL_VOID_STATE_READY'

    errors.push('In-memory spatial unit contamination survived Tabula Rasa Reset')

    `>> [SENTINEL:PARITY] Spatial round-trip Void-State parity certified [${valid ? 'PASS' : 'FAIL'}]`,
  ]

    sandboxRoot: resolvedSandbox,

    postResetVoidVerified,
    zeroOrphanFilesVerified,
    zeroModelContaminationVerified,
    errors,
    receipts,
  }
}
