export interface SpatialResetGuardOptions {
  readonly allowDownstreamSibling?: boolean
  readonly engineRoot?: string
}

export function assertSafeSpatialResetTarget(
  repoRoot: string,
  candidatePath: string,
  options: SpatialResetGuardOptions = {},
): string {
  if (typeof repoRoot !== 'string' || repoRoot.trim().length === 0) {
    throw new Error('[UNSAFE_SPATIAL_RESET_PATH] repoRoot must be non-empty')
  }
  if (typeof candidatePath !== 'string' || candidatePath.trim().length === 0) {
    throw new Error(
      '[UNSAFE_SPATIAL_RESET_PATH] candidatePath must be non-empty',
    )
  }

  const resolvedRoot = path.resolve(repoRoot)
  const resolvedTarget = path.isAbsolute(candidatePath)
    ? path.resolve(candidatePath)
    : path.resolve(resolvedRoot, candidatePath)

  if (resolvedTarget === resolvedRoot) {
    throw new Error(
      '[UNSAFE_SPATIAL_RESET_PATH] Refusing to purge repository root itself',
    )
  }

  const relativeFromRoot = path.relative(resolvedRoot, resolvedTarget)
  const isOutsideRoot =
    relativeFromRoot.startsWith('..') || path.isAbsolute(relativeFromRoot)

  if (isOutsideRoot) {
    if (options.allowDownstreamSibling) {
      const expectedEngineRoot = options.engineRoot
        ? path.resolve(options.engineRoot)
        : path.resolve(resolvedRoot, '..', '000.series-sower-engine')
      const expectedSpatialInbox = path.join(
        expectedEngineRoot,
        'data',
        'spatial',
      )
      if (resolvedTarget === expectedSpatialInbox) {
        return resolvedTarget
      }
    }
    throw new Error(
      `[UNSAFE_SPATIAL_RESET_PATH] Target escapes spatial root boundary: ${candidatePath}`,
    )
  }

  const normalizedRel = relativeFromRoot.replace(/\\/g, '/')
  const segments = normalizedRel.split('/').filter(Boolean)

  for (const protectedEntry of SPATIAL_PROTECTED_PATHS) {
    const cleanProtected = protectedEntry.replace(/\\/g, '/')
    if (
      normalizedRel === cleanProtected ||
      normalizedRel.startsWith(`${cleanProtected}/`) ||
      segments[0] === cleanProtected
    ) {
      throw new Error(
        `[UNSAFE_SPATIAL_RESET_PATH] Target intersects protected path [${cleanProtected}]: ${normalizedRel}`,
      )
    }
  }

  const basename = path.basename(resolvedTarget)
  if (basename.startsWith('.env') || basename.startsWith('.dev.vars')) {
    throw new Error(
      `[UNSAFE_SPATIAL_RESET_PATH] Refusing to purge secret file: ${basename}`,
    )
  }

  return resolvedTarget
}

export interface SpatialPerimeterAuditResult {
  readonly ok: boolean
  readonly verifiedTargets: readonly string[]
  readonly statefulUnitCount: number
  readonly statelessUnitCount: number
  readonly receipts: readonly string[]
}

export async function verifySpatialResetPerimeter(
  repoRoot: string,
  ste?: State,
  engineRoot?: string,
): Promise<SpatialPerimeterAuditResult> {
  const manifest = validateSpatialResetManifest(SPATIAL_RESET_MANIFEST)
  const verifiedTargets: string[] = []
  const receipts: string[] = [
    '>> ================================================================',
    '>> [SPATIAL:RESET_GUARD] Auditing Declarative Spatial Reset Manifest',
  ]

  for (const descriptor of manifest.artifactTargets) {
    const safePath = assertSafeSpatialResetTarget(
      repoRoot,
      descriptor.relativePath,
      {
        allowDownstreamSibling: descriptor.allowDownstreamSibling,
        engineRoot,
      },
    )
    verifiedTargets.push(safePath)
    receipts.push(
      `>> [RESET:TARGET] [${descriptor.durabilityClass}] ${descriptor.relativePath} -> VERIFIED`,
    )
  }

  const statefulUnitCount = manifest.unitRegistry.filter(
    (u) => u.classification === 'STATEFUL_RESETTABLE',
  ).length
  const statelessUnitCount = manifest.unitRegistry.filter(
    (u) => u.classification === 'STATELESS_PURE',
  ).length

  receipts.push(
    `>> [RESET:UNITS] Total=${manifest.unitRegistry.length} | Stateful=${statefulUnitCount} | Stateless=${statelessUnitCount}`,
  )
  receipts.push(
    '>> [RESET:GUARD] Spatial Tabula Rasa perimeter verified [OK]',
  )
  receipts.push(
    '>> ================================================================',
  )

  for (const line of receipts) {
    await emitSpatialTelemetry(ste, line)
  }

  return {
    ok: true,
    verifiedTargets,
    statefulUnitCount,
    statelessUnitCount,
    receipts,
  }
}
