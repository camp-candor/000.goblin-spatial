export interface DownstreamSpatialPurgeResult {
  readonly purgedFiles: number
  readonly purgedSeriesDirs: number
  readonly targetSpatialDir: string
}

export async function purgeDownstreamSpatialProofs(options?: {
  readonly spatialRoot?: string
  readonly engineRoot?: string
  readonly ste?: State
}): Promise<DownstreamSpatialPurgeResult> {
  const spatialRoot = options?.spatialRoot
    ? path.resolve(options.spatialRoot)
    : path.resolve(process.cwd())
  const engineRoot = options?.engineRoot
    ? path.resolve(options.engineRoot)
    : resolveDefaultEngineRoot(spatialRoot)

  const candidateDir = path.join(engineRoot, 'data', 'spatial')
  const safeSpatialDir = assertSafeSpatialResetTarget(
    spatialRoot,
    candidateDir,
    {
      allowDownstreamSibling: true,
      engineRoot,
    },
  )

  let purgedFiles = 0
  let purgedSeriesDirs = 0

  const countFilesRecursively = (dir: string): number => {
    if (!fs.existsSync(dir)) return 0
    let count = 0
    for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
      if (entry.name === '.gitkeep') continue
      const fullPath = path.join(dir, entry.name)
      if (entry.isDirectory()) {
        count += countFilesRecursively(fullPath)
      } else if (entry.isFile()) {
        count++
      }
    }
    return count
  }

  if (fs.existsSync(safeSpatialDir)) {
    const entries = fs.readdirSync(safeSpatialDir, { withFileTypes: true })
    for (const entry of entries) {
      if (entry.name === '.gitkeep') continue
      const childPath = path.join(safeSpatialDir, entry.name)
      const safeChild = assertSafeSpatialResetTarget(spatialRoot, childPath, {
        allowDownstreamSibling: true,
        engineRoot,
      })
      if (entry.isDirectory()) {
        purgedFiles += countFilesRecursively(safeChild)
        purgedSeriesDirs++
      } else if (entry.isFile()) {
        purgedFiles++
      }
      fs.rmSync(safeChild, { recursive: true, force: true })
    }
  }

  const line = `>> [SPATIAL:DOWNSTREAM_PURGE] Purged ${purgedFiles} proof file(s) across ${purgedSeriesDirs} engine series folder(s) [OK]`
  const lib = (globalThis as any).LIBRARY || (global as any).LIBRARY
  if (lib && typeof lib.hunt === 'function') {
    await lib
      .hunt('[Console action] Update Console', { idx: 'cns00', src: line })
      .catch(() => {})
  } else if (options?.ste && typeof options.ste.hunt === 'function') {
    await options.ste
      .hunt('[Console action] Update Console', { idx: 'cns00', src: line })
      .catch(() => {})
  }

  return {
    purgedFiles,
    purgedSeriesDirs,
    targetSpatialDir: safeSpatialDir,
  }
}

export function resolveResetSpatialRoot(startDir?: string): string {
  if (startDir && startDir.trim().length > 0) {
    return path.resolve(startDir)
  }
  let curr = path.resolve(process.cwd())
  while (curr && curr !== path.dirname(curr)) {
    if (
      fs.existsSync(path.join(curr, 'packages', '301.spatial')) ||
      fs.existsSync(path.join(curr, '00.spatial.unit'))
    ) {
      return curr
    }
    curr = path.dirname(curr)
  }
  return path.resolve(process.cwd())
}

function countFilesInDir(dir: string, extFilter?: string): number {
  if (!fs.existsSync(dir)) return 0
  let total = 0
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    if (entry.name === '.gitkeep') continue
    const fullPath = path.join(dir, entry.name)
    if (entry.isDirectory()) {
      total += countFilesInDir(fullPath, extFilter)
    } else if (entry.isFile()) {
      if (!extFilter || entry.name.endsWith(extFilter)) {
        total++
      }
    }
  }
  return total
}

function countScratchpadsInDir(dir: string): number {
  if (!fs.existsSync(dir)) return 0
  let total = 0
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    const fullPath = path.join(dir, entry.name)
    if (entry.isDirectory()) {
      if (
        entry.name === 'node_modules' ||
        entry.name === '.git' ||
        entry.name === 'dist'
      ) {
        continue
      }
      total += countScratchpadsInDir(fullPath)
    } else if (entry.isFile()) {
      if (
        entry.name.startsWith('.tmp_') ||
        entry.name.includes('.tmp_') ||
        entry.name.endsWith('.tmp')
      ) {
        total++
      }
    }
  }
  return total
}

export async function scanSpatialBlastRadius(
  repoRootInput?: string,
  ste?: State,
  engineRootInput?: string,
): Promise<SpatialBlastRadiusInventory> {
  const repoRoot = resolveResetSpatialRoot(repoRootInput)
  const engineRoot = engineRootInput
    ? path.resolve(engineRootInput)
    : resolveDefaultEngineRoot(repoRoot)

  const receiptsDir = path.join(repoRoot, 'spatial_receipts')
  const manifestsDir = path.join(repoRoot, 'data', 'manifests')
  const localProofsDir = path.join(repoRoot, 'data', 'spatial_proofs')
  const stageDraftsDir = path.join(repoRoot, 'drafts', 'spatial_stage')
  const downstreamSpatialDir = path.join(engineRoot, 'data', 'spatial')

  const signedReceiptPackets = countFilesInDir(receiptsDir, '.json')
  const cachedTopologyManifests = countFilesInDir(manifestsDir, '.json')
  const stagedAppointmentProofs =
    countFilesInDir(localProofsDir, '.json') +
    countFilesInDir(stageDraftsDir, '.json')
  const downstreamEngineProofFiles = countFilesInDir(
    downstreamSpatialDir,
    '.json',
  )

  const tempScratchpadFiles =
    countScratchpadsInDir(receiptsDir) +
    countScratchpadsInDir(manifestsDir) +
    countScratchpadsInDir(localProofsDir) +
    countScratchpadsInDir(stageDraftsDir) +
    countScratchpadsInDir(path.join(repoRoot, 'data'))

  const stateVal = (ste as any)?.value || {}
  const loadedHexCells =
    typeof stateVal.topology?.totalLoadedCells === 'number'
      ? stateVal.topology.totalLoadedCells
      : typeof stateVal.database?.topologyDb?.size === 'function'
        ? stateVal.database.topologyDb.size()
        : 0
  const registeredPartitions =
    typeof stateVal.edge?.registeredPartitionCount === 'number'
      ? stateVal.edge.registeredPartitionCount
      : 0
  const ledgerQuadrupleEntries = Array.isArray(stateVal.ledger?.entries)
    ? stateVal.ledger.entries.length
    : 0

  const inventory: SpatialBlastRadiusInventory = {
    cachedTopologyManifests,
    signedReceiptPackets,
    stagedAppointmentProofs,
    downstreamEngineProofFiles,
    loadedHexCells,
    registeredPartitions,
    ledgerQuadrupleEntries,
    tempScratchpadFiles,
  }

  const telemetryLines: readonly string[] = [
    '>> ================================================================',
    '>> [SPATIAL:BLAST_RADIUS] PRE-FLIGHT VOID-STATE INVENTORY SWEEP',
    `>> Cached Manifests  : ${inventory.cachedTopologyManifests} topology manifest(s)`,
    `>> Signed Receipts   : ${inventory.signedReceiptPackets} receipt packet(s)`,
    `>> Staged Proofs     : ${inventory.stagedAppointmentProofs} local | ${inventory.downstreamEngineProofFiles} engine handoff`,
    `>> Scratchpad Files  : ${inventory.tempScratchpadFiles} temporary buffer(s)`,
    `>> Live Heap State   : Cells=${inventory.loadedHexCells} | Partitions=${inventory.registeredPartitions} | LedgerEntries=${inventory.ledgerQuadrupleEntries}`,
    '>> ================================================================',
  ]
  await emitSpatialResetTelemetry(ste, telemetryLines)

  return inventory
}

function purgeDirectoryContentsSafe(
  repoRoot: string,
  targetDir: string,
  ensureContainer: boolean,
): number {
  const safeDir = assertSafeSpatialResetTarget(repoRoot, targetDir)
  let removedFiles = 0

  if (fs.existsSync(safeDir)) {
    const entries = fs.readdirSync(safeDir, { withFileTypes: true })
    for (const entry of entries) {
      if (entry.name === '.gitkeep') continue
      const childPath = path.join(safeDir, entry.name)
      const safeChild = assertSafeSpatialResetTarget(repoRoot, childPath)
      if (entry.isDirectory()) {
        removedFiles += countFilesInDir(safeChild)
      } else if (entry.isFile()) {
        removedFiles++
      }
      fs.rmSync(safeChild, { recursive: true, force: true })
    }
  } else if (ensureContainer) {
    fs.mkdirSync(safeDir, { recursive: true })
  }

  return removedFiles
}

function purgeOrphanedScratchpadsSafe(repoRoot: string): number {
  const scanRoots = [
    path.join(repoRoot, 'spatial_receipts'),
    path.join(repoRoot, 'data'),
    path.join(repoRoot, 'drafts'),
  ]
  let removed = 0

  const sweep = (currentDir: string): void => {
    if (!fs.existsSync(currentDir)) return
    for (const entry of fs.readdirSync(currentDir, { withFileTypes: true })) {
      const fullPath = path.join(currentDir, entry.name)
      if (entry.isDirectory()) {
        if (entry.name === 'directive' || entry.name === 'overview' || entry.name === 'building') {
          continue
        }
        sweep(fullPath)
      } else if (entry.isFile()) {
        if (
          entry.name.startsWith('.tmp_') ||
          entry.name.includes('.tmp_') ||
          entry.name.endsWith('.tmp')
        ) {
          const safeFile = assertSafeSpatialResetTarget(repoRoot, fullPath)
          fs.rmSync(safeFile, { force: true })
          removed++
        }
      }
    }
  }

  for (const root of scanRoots) {
    sweep(root)
  }
  return removed
}

export interface SpatialVoidVerificationResult {
  readonly isVoidState: boolean
  readonly residualDiskArtifacts: number
  readonly ledgerHeadHash: string
  readonly loadedManifestCount: number
}

export async function verifySpatialVoidState(
  repoRootInput?: string,
  ste?: State,
  engineRootInput?: string,
): Promise<SpatialVoidVerificationResult> {
  const postInventory = await scanSpatialBlastRadius(
    repoRootInput,
    undefined,
    engineRootInput,
  )
  const residualDiskArtifacts =
    postInventory.cachedTopologyManifests +
    postInventory.signedReceiptPackets +
    postInventory.stagedAppointmentProofs +
    postInventory.downstreamEngineProofFiles +
    postInventory.tempScratchpadFiles

  const stateVal = (ste as any)?.value || {}
  const ledgerHeadHash =
    typeof stateVal.ledger?.lastEventHash === 'string'
      ? stateVal.ledger.lastEventHash
      : GENESIS_HASH_64
  const loadedManifestCount =
    typeof stateVal.database?.topologyDb?.size === 'function'
      ? stateVal.database.topologyDb.size()
      : 0

  if (stateVal.menu && typeof stateVal.menu === 'object') {
    stateVal.menu.statusMessage = 'SPATIAL_VOID_STATE_READY'
    stateVal.menu.lastAuditSummary = null
  }

  const isVoidState =
    residualDiskArtifacts === 0 &&
    loadedManifestCount === 0 &&
    ledgerHeadHash === GENESIS_HASH_64

  await emitSpatialResetTelemetry(ste, [
    `>> [SPATIAL:VOID_VERIFIED] isVoidState=${isVoidState} | ResidualArtifacts=${residualDiskArtifacts} | LedgerHead=${ledgerHeadHash.slice(0, 12)}... [OK]`,
  ])

  return {
    isVoidState,
    residualDiskArtifacts,
    ledgerHeadHash,
    loadedManifestCount,
  }
}

export const executeTabulaRasaSpatial = async (
  cpy: SpatialModel,
  bal?: SpatialBit,
  ste?: State,
): Promise<SpatialModel> => {
  const datObj = (bal?.dat || {}) as Record<string, any>
  const req: SpatialTabulaRasaRequestBit =
    (datObj.tabulaRasaRequest as SpatialTabulaRasaRequestBit) || datObj

  const repoRoot = resolveResetSpatialRoot(
    req.repoRoot || bal?.src || process.cwd(),
  )
  const engineRoot = req.engineRoot
    ? path.resolve(req.engineRoot)
    : resolveDefaultEngineRoot(repoRoot)

    req.cascadeOrigin || 'LOCAL_TUI'

    cachedTopologyManifests: 0,
    signedReceiptPackets: 0,
    stagedAppointmentProofs: 0,
    downstreamEngineProofFiles: 0,
    loadedHexCells: 0,
    registeredPartitions: 0,
    ledgerQuadrupleEntries: 0,
    tempScratchpadFiles: 0,
  }

  let abortStage: 1 | 2 | 3 | undefined

    abortStage = 1
  } else if (req.confirmToken2 !== SPATIAL_TABULA_RASA_TOKEN_2) {
    abortStage = 2
  } else if (req.confirmToken3 !== SPATIAL_TABULA_RASA_TOKEN_3) {
    abortStage = 3
  }

  if (abortStage !== undefined) {
    const abortLine = `:: [SPATIAL:TABULA_RASA_ABORTED] Triple-Lock verification failed at Stage ${abortStage}/3. Workspace untouched.`
    await emitSpatialResetTelemetry(ste, [abortLine])

    const abortedReceipt: SpatialTabulaRasaReceipt = {
      ok: false,
      voidStateVerified: false,
      cascadeOrigin,
      purgedDiskCounts: zeroInventory,
      resetUnitIds: [],
      genesisLedgerHeadHash: GENESIS_HASH_64,
      timestampIso: new Date().toISOString(),
      receipts: [abortLine],
    }

    if (bal?.slv) {
      bal.slv({
        sptBit: {
          idx: 'execute-tabula-rasa-spatial-aborted',
          val: 0,
          dat: abortedReceipt,
        } as any,
      })
    }
    return cpy
  }

  const preInventory = await scanSpatialBlastRadius(repoRoot, ste, engineRoot)

    `>> [SPATIAL:PURGE_DISK] Initiating Tabula Rasa purge (origin=${cascadeOrigin})...`,
  ]

    repoRoot,
    path.join(repoRoot, 'spatial_receipts'),
    true,
  )

    repoRoot,
    path.join(repoRoot, 'data', 'manifests'),
    true,
  )
  const purgedLocalProofs = purgeDirectoryContentsSafe(
    repoRoot,
    path.join(repoRoot, 'data', 'spatial_proofs'),
    true,
  )
  const purgedStageDrafts = purgeDirectoryContentsSafe(
    repoRoot,
    path.join(repoRoot, 'drafts', 'spatial_stage'),
    false,
  )
  const purgedScratchpads = purgeOrphanedScratchpadsSafe(repoRoot)

  const downstreamPurge = await purgeDownstreamSpatialProofs({
    spatialRoot: repoRoot,
    engineRoot,
    ste,
  })

  receipts.push(
    `>> [SPATIAL:PURGE_DISK] Receipts=${purgedReceipts} | Manifests=${purgedManifests} | LocalProofs=${purgedLocalProofs + purgedStageDrafts} | Scratchpads=${purgedScratchpads} [OK]`,
  )

  // Zero local SpatialModel counters
  cpy.totalSynchronousInvocations = 0
  cpy.totalBudgetsTripped = 0
  cpy.lastExecutionMicros = 0

  // Dispatch Phase 2 RESET_<UNIT> actions across all 16 STATEFUL_RESETTABLE units
  const resetUnitIds: string[] = []
  for (const unitDesc of SPATIAL_RESET_MANIFEST.unitRegistry) {
    if (
      unitDesc.classification === 'STATEFUL_RESETTABLE' &&
      unitDesc.resetActionConstant
    ) {
      resetUnitIds.push(unitDesc.unitFolder)
      if (
        unitDesc.unitFolder !== '00.spatial.unit' &&
        ste &&
        typeof ste.hunt === 'function'
      ) {
        await ste.hunt(unitDesc.resetActionConstant, {}).catch(() => {})
      }
    }
  }

  receipts.push(
    `>> [SPATIAL:ZERO_UNITS] Re-initialized ${resetUnitIds.length} stateful spatial unit(s) -> VOID STATE [OK]`,
  )

  const voidCheck = await verifySpatialVoidState(repoRoot, ste, engineRoot)
  receipts.push(
    `>> [TABULA_RASA:COMPLETE] 000.goblin-spatial restored to Byte-0 Void State [PASS]`,
  )

  await emitSpatialResetTelemetry(ste, receipts)

  const finalReceipt: SpatialTabulaRasaReceipt = {
    ok: voidCheck.isVoidState,
    voidStateVerified: voidCheck.isVoidState,
    cascadeOrigin,
    purgedDiskCounts: {
      cachedTopologyManifests: purgedManifests,
      signedReceiptPackets: purgedReceipts,
      stagedAppointmentProofs: purgedLocalProofs + purgedStageDrafts,
      downstreamEngineProofFiles: downstreamPurge.purgedFiles,
      loadedHexCells: preInventory.loadedHexCells,
      registeredPartitions: preInventory.registeredPartitions,
      ledgerQuadrupleEntries: preInventory.ledgerQuadrupleEntries,
      tempScratchpadFiles: purgedScratchpads,
    },
    resetUnitIds,
    genesisLedgerHeadHash: voidCheck.ledgerHeadHash,
    timestampIso: new Date().toISOString(),
    receipts,
  }

  if (bal?.slv) {
    bal.slv({
      sptBit: {
        idx: 'execute-tabula-rasa-spatial-pass',
        val: voidCheck.isVoidState ? 1 : 0,
        dat: finalReceipt,
      } as any,
    })
  }

  return cpy
}
